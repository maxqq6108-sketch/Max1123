--────────────────────────────── Слой файловой системы ────────────────
-- Универсальная обёртка: любой экзекутор без writefile/listfiles не сломает скрипт.
local FS = { ok = false, caps = {}, memFiles = {}, memWarned = false }
Nova.FS = FS

local function tryCap(name, fn)
	local ok = pcall(fn)
	FS.caps[name] = ok
	return ok
end

-- Многие экзекуторы требуют существующую папку: создаём корень до пробы
pcall(function() makefolder("NovaMusicData") end)

FS.ok = tryCap("writefile", function() writefile("NovaMusicData/.probe", "1") end)
if FS.ok then
	FS.caps.readfile  = tryCap("readfile",  function() readfile("NovaMusicData/.probe") end)
	FS.caps.isfile    = tryCap("isfile",    function() isfile("NovaMusicData/.probe") end)
	FS.caps.isfolder  = tryCap("isfolder",  function() isfolder("NovaMusicData") end)
	FS.caps.makefolder= tryCap("makefolder",function() end)
	FS.caps.listfiles = tryCap("listfiles", function() listfiles("NovaMusicData") end)
	FS.caps.delfile   = tryCap("delfile",   function() delfile("NovaMusicData/.probe") end)
else
	-- Полностью in-memory режим (экстремальный fallback)
	FS.caps.readfile, FS.caps.isfile, FS.caps.isfolder = true, true, true
	FS.caps.makefolder, FS.caps.listfiles, FS.caps.delfile = true, true, true
	function FS.memWarn()
		if not FS.memWarned then
			FS.memWarned = true
			warn("[NovaMusic] Файловая система недоступна — работаю в памяти. Конфиг и библиотека не сохранятся между сессиями.")
		end
	end
end

function FS.isFile(path)
	if not FS.ok then return FS.memFiles[path] ~= nil end
	local ok, res = pcall(isfile, path)
	return ok and res or false
end

function FS.isFolder(path)
	if not FS.ok then return false end
	local ok, res = pcall(isfolder, path)
	return ok and res or false
end

function FS.makeFolder(path)
	if not FS.ok then return false end
	local ok = pcall(makefolder, path)
	return ok
end

function FS.ensureFolder(path)
	if FS.isFolder(path) then return true end
	return FS.makeFolder(path)
end

function FS.read(path)
	if not FS.ok then return FS.memFiles[path] end
	local ok, res = pcall(readfile, path)
	if ok then return res end
	return nil
end

function FS.write(path, data)
	if not FS.ok then
		FS.memWarn()
		FS.memFiles[path] = data
		return true
	end
	return pcall(writefile, path, data)
end

function FS.delete(path)
	if not FS.ok then
		FS.memFiles[path] = nil
		return true
	end
	return pcall(delfile, path)
end

function FS.list(folder)
	if not FS.ok then
		local out = {}
		for p in pairs(FS.memFiles) do
			if p:sub(1, #folder + 1) == folder .. "/" and not p:find("/", #folder + 2, true) then
				out[#out + 1] = p
			end
		end
		return out
	end
	local ok, res = pcall(listfiles, folder)
	if ok and type(res) == "table" then return res end
	return {}
end

function U.clamp(v, a, b) if v < a then return a elseif v > b then return b end return v end
function U.lerp(a, b, t) return a + (b - a) * t end
function U.round(v, dec)
	local m = 10 ^ (dec or 0)
	return math.floor(v * m + 0.5) / m
end

-- Формат времени mm:ss
function U.fmtTime(sec)
	if type(sec) ~= "number" or sec ~= sec or sec < 0 then return "0:00" end
	local m = math.floor(sec / 60)
	local s = math.floor(sec % 60)
	return string.format("%d:%02d", m, s)
end

-- FNV-1a хеш строки → стабильное число (для сидов обложек/лирики)
function U.hash(str)
	local h = 2166136261
	str = tostring(str or "")
	for i = 1, #str do
		h = (h * 16777619 + string.byte(str, i)) % 4294967296
	end
	return h
end

function U.seededRandom(seed)
	local s = (tonumber(seed) or 1) % 2147483647
	if s <= 0 then s = s + 2147483646 end
	return function(min, max)
		s = (s * 16807) % 2147483647
		if min and max then return min + (s / 2147483647) * (max - min) end
		return s / 2147483647
	end
end

function U.deepCopy(t)
	if type(t) ~= "table" then return t end
	local r = {}
	for k, v in pairs(t) do r[k] = U.deepCopy(v) end
	return r
end

function U.deepMerge(base, override)
	local r = U.deepCopy(base)
	if type(override) == "table" then
		for k, v in pairs(override) do
			if type(v) == "table" and type(r[k]) == "table" then
				r[k] = U.deepMerge(r[k], v)
			else
				r[k] = U.deepCopy(v)
			end
		end
	end
	return r
end

function U.indexOf(arr, val)
	for i = 1, #arr do if arr[i] == val then return i end end
	return nil
end

function U.uid()
	Nova.uid = Nova.uid + 1
	return Nova.uid
end

-- Ограничитель частоты вызовов
function U.debounce(sec, fn)
	local last = 0
	return function(...)
		local now = os.clock()
		if now - last >= sec then
			last = now
			return fn(...)
		end
	end
end

-- Lower с поддержкой кириллицы (string.lower понимает только ASCII)
function U.lowerRu(str)
	str = tostring(str or ""):lower() -- ASCII-часть
	local out = {}
	local i, n = 1, #str
	while i <= n do
		local b1 = str:byte(i)
		local b2 = str:byte(i + 1)
		if b1 == 0xD0 and b2 then
			if b2 == 0x81 then
				-- Ё (D0 81) → ё (D1 91)
				out[#out + 1] = "\xD1\x91"
				i = i + 2
			elseif b2 >= 0x90 and b2 <= 0x9F then
				-- А..П (D0 90..9F) → а..п (D0 B0..BF)
				out[#out + 1] = string.char(0xD0, b2 + 0x20)
				i = i + 2
			elseif b2 >= 0xA0 and b2 <= 0xAF then
				-- Р..Я (D0 A0..AF) → р..я (D1 80..8F)
				out[#out + 1] = string.char(0xD1, b2 - 0x20)
				i = i + 2
			else
				out[#out + 1] = string.char(b1, b2)
				i = i + 2
			end
		else
			out[#out + 1] = string.char(b1)
			i = i + 1
		end
	end
	return table.concat(out)
end

-- Отложенный вызов с перезапуском (сохранение конфига)
function U.idle(sec, fn)
	local token = { alive = true }
	local function schedule(...)
		local args = { ... }
		token.alive = false
		local newToken = { alive = true }
		task.delay(sec, function()
			if newToken.alive then fn(unpack(args)) end
		end)
		return newToken
	end
	local f
	f = function(...)
		token = schedule(...)
	end
	return f
end

--────────────────────────────── Сигналы ──────────────────────────────
local Signal = {}
Signal.__index = Signal

function Signal.new()
	return setmetatable({ _handlers = {} }, Signal)
end

function Signal:Connect(fn)
	local handlers = self._handlers
	local entry = { fn = fn, once = false }
	handlers[#handlers + 1] = entry
	return {
		Disconnect = function()
			for i, h in ipairs(handlers) do
				if h == entry then table.remove(handlers, i) break end
			end
		end
	}
end

function Signal:Once(fn)
	local handlers = self._handlers
	local entry = { fn = fn, once = true }
	handlers[#handlers + 1] = entry
	return {
		Disconnect = function()
			for i, h in ipairs(handlers) do
				if h == entry then table.remove(handlers, i) break end
			end
		end
	}
end

function Signal:Fire(...)
	local nargs = select("#", ...)
	local args = { ... }
	-- снимок: once-обработчики должны сработать, но остаться в списке до вызова
	local snapshot = {}
	for i, h in ipairs(self._handlers) do snapshot[i] = h end
	for i = #self._handlers, 1, -1 do
		if self._handlers[i].once then table.remove(self._handlers, i) end
	end
	for _, h in ipairs(snapshot) do
		task.spawn(h.fn, unpack(args, 1, nargs))
	end
end

function Signal:Destroy()
	self._handlers = {}
end

Nova.Signal = Signal

local Lib = { tracks = {}, byKey = {}, MAX_TRACKS = 200 }

function Lib.likedKey(track)
	if track.demo then return "demo:" .. track.title end
	if track.id then return tostring(track.id) end
	return "name:" .. track.title
end

function Lib.isLiked(track)
	local k = Lib.likedKey(track)
	for _, v in ipairs(Config.data.liked) do
		if v == k then return true end
	end
	return false
end

function Lib.toggleLiked(track)
	local k = Lib.likedKey(track)
	local found = false
	for i, v in ipairs(Config.data.liked) do
		if v == k then table.remove(Config.data.liked, i) found = true break end
	end
	if not found then
		table.insert(Config.data.liked, k)
	end
	Config.save()
	return not found
end

function Lib.normalize(data, fileName)
	if type(data) ~= "table" then return nil end
	local title = tostring(data.title or data.name or fileName or "Без названия")
	local artist = tostring(data.artist or data.author or "Неизвестный исполнитель")
	local id = nil
	if type(data.id) == "number" and data.id > 0 then
		id = "rbxassetid://" .. tostring(data.id)
	elseif type(data.id) == "string" and #data.id > 4 then
		if data.id:match("^rbxasset") or data.id:match("^http") then
			id = data.id
		elseif tonumber(data.id) then
			id = "rbxassetid://" .. data.id
		end
	end
	local track = {
		title    = title,
		artist   = artist,
		id       = id,
		demo     = (not id) or (data.demo == true),
		duration = U.clamp(tonumber(data.duration) or 200, 10, 3600),
		bpm      = U.clamp(tonumber(data.bpm) or 104, 40, 220),
		lyrics   = nil,
		file     = fileName,
		key      = nil,
	}
	if type(data.lyrics) == "table" and #data.lyrics > 0 then
		local lines = {}
		for _, l in ipairs(data.lyrics) do
			if type(l) == "table" and type(l.line) == "string" then
				lines[#lines + 1] = { t = tonumber(l.t) or 0, line = l.line }
			end
		end
		table.sort(lines, function(a, b) return a.t < b.t end)
		if #lines > 0 then track.lyrics = lines end
	end
	track.key = Lib.likedKey(track)
	return track
end

function Lib.scan()
	local files = FS.list(SONGS)
	local tracks = {}
	for _, path in ipairs(files) do
		local name = path:match("[^/\\]+$") or path
		local lower = name:lower()
		if name:sub(1, 1) ~= "_" and (lower:match("%.json$") or lower:match("%.nova$")) then
			local raw = FS.read(path)
			local data = Nova.jsonDecode(raw)
			local t = Lib.normalize(data, name:gsub("%.json$", ""):gsub("%.nova$", ""))
			if t then
				tracks[#tracks + 1] = t
				if #tracks >= Lib.MAX_TRACKS then break end
			end
		end
	end
	-- Нет валидных треков → создаём демо-записи и пересканируем
	if #tracks == 0 then
		Lib.writeDemoSongs()
		for _, path in ipairs(FS.list(SONGS)) do
			local name = path:match("[^/\\]+$") or path
			local lower = name:lower()
			if name:sub(1, 1) ~= "_" and lower:match("%.json$") then
				local data = Nova.jsonDecode(FS.read(path))
				local t = Lib.normalize(data, name:gsub("%.json$", ""))
				if t then tracks[#tracks + 1] = t end
			end
		end
	end
	table.sort(tracks, function(a, b)
		if a.demo ~= b.demo then return not a.demo end -- реальные треки выше
		return a.title:lower() < b.title:lower()
	end)
	for i, t in ipairs(tracks) do t.index = i end
	Lib.tracks = tracks
	Lib.byKey = {}
	for _, t in ipairs(tracks) do Lib.byKey[t.key] = t end
	bus.LibChanged:Fire(tracks)
	return tracks
end

function Lib.findByKey(key)
	return Lib.byKey[key]
end

function Lib.init()
	ensureTree()
	Config.load()
	-- config.json появляется сразу при первом запуске
	if not FS.isFile(ROOT .. "/config.json") then
		Config.flush()
	end
	Lib.scan()
end

--══════════════════════════ Часть 4: NVG — векторный движок ══════════════════════════
-- Nova Vector Graphics: программный растеризатор примитивов. Никаких эмодзи и
-- готовых картинок — все иконки рисуются кадрами (Frame + UICorner + UIGradient):
--   rect    — прямоугольник со скруглением
--   circle  — круг (UICorner 1,0)
--   line    — капсульная линия с поворотом
--   ring    — дуга/кольцо из сегментов
--   poly    — заливка полигона построчным сканлайном
--   polyline/arrow — ломаные и стрелки
-- Система координат иконок — расчётная сетка 24×24, масштабируется в любой px.

local U = Nova.util
local Theme = Nova.theme

local NVG = {}
Nova.nvg = NVG

-- Переносимый atan2 (в Luau есть math.atan2, но не во всех Lua-окружениях)
local function atan2(y, x)
	if x > 0 then return math.atan(y / x) end
	if x < 0 then
		if y >= 0 then return math.atan(y / x) + math.pi end
		return math.atan(y / x) - math.pi
	end
	if y > 0 then return math.pi / 2 end
	if y < 0 then return -math.pi / 2 end
	return 0
end

--────────────────────────── Перо (контекст рисования) ──────────────────────────
local Pen = {}
Pen.__index = Pen

-- Создаёт контекст: координаты художника (design×design) масштабируются в pxSize
function NVG.pen(host, pxSize, design)
	local pen = setmetatable({
		host = host,
		s = pxSize / (design or 24),
		color = Theme.colors.white,
		alpha = 0,
		bg = Theme.colors.bg,     -- цвет «прорезки» (knockout)
		g1 = nil, g2 = nil, grot = 0,
		count = 0,
	}, Pen)
	return pen
end

function Pen:st(opts)
	if opts.color ~= nil then self.color = opts.color end
	if opts.alpha ~= nil then self.alpha = opts.alpha end
	if opts.bg ~= nil then self.bg = opts.bg end
	if opts.grad ~= nil then
		self.g1, self.g2, self.grot = opts.grad[1], opts.grad[2], opts.grad[3] or 90
	elseif opts.grad == false then
		self.g1, self.g2 = nil, nil
	end
	return self
end

local function applyStyle(frame, pen)
	frame.BackgroundColor3 = pen.color
	frame.BackgroundTransparency = pen.alpha
	frame.BorderSizePixel = 0
	frame.Parent = pen.host
	pen.count = pen.count + 1
	if pen.g1 then
		local g = Instance.new("UIGradient")
		g.Color = ColorSequence.new(pen.g1, pen.g2)
		g.Rotation = pen.grot
		g.Parent = frame
	end
	return frame
end

-- Прямоугольник: левый-верхний угол (x,y), размеры (w,h), скругление r
function Pen:rect(x, y, w, h, r)
	local f = Instance.new("Frame")
	f.AnchorPoint = Vector2.new(0, 0)
	f.Position = UDim2.fromOffset(x * self.s, y * self.s)
	f.Size = UDim2.fromOffset(math.max(w * self.s, 1), math.max(h * self.s, 1))
	applyStyle(f, self)
	if r and r > 0 then
		local c = Instance.new("UICorner")
		c.CornerRadius = UDim.new(0, r * self.s)
		c.Parent = f
	end
	return f
end

-- Круг: центр (cx,cy), диаметр d
function Pen:circle(cx, cy, d)
	local r = d * self.s / 2
	local f = Instance.new("Frame")
	f.AnchorPoint = Vector2.new(0.5, 0.5)
	f.Position = UDim2.fromOffset(cx * self.s, cy * self.s)
	f.Size = UDim2.fromOffset(math.max(r * 2, 1), math.max(r * 2, 1))
	applyStyle(f, self)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(1, 0)
	c.Parent = f
	return f
end

-- Круг «прорезки» цветом фона (для отверстий в шестерёнке, точках слайдеров)
function Pen:hole(cx, cy, d)
	local saved = self.color
	self.color = self.bg
	self:circle(cx, cy, d)
	self.color = saved
end

-- Прямоугольная прорезка цветом фона
function Pen:knockRect(x, y, w, h, r)
	local saved = self.color
	self.color = self.bg
	self:rect(x, y, w, h, r)
	self.color = saved
end

-- Контурное сердце: заливка + прорезка меньшим сердцем
function Pen:heartOutline(cx, cy, scale)
	self:heart(cx, cy, scale)
	local saved = self.color
	self.color = self.bg
	self:heart(cx, cy - 0.6, scale * 0.74)
	self.color = saved
end

-- Линия: капсула между точками, толщина th
function Pen:line(x1, y1, x2, y2, th)
	local dx, dy = (x2 - x1) * self.s, (y2 - y1) * self.s
	local len = math.sqrt(dx * dx + dy * dy)
	local ang = math.deg(atan2(dy, dx))
	local f = Instance.new("Frame")
	f.AnchorPoint = Vector2.new(0.5, 0.5)
	f.Position = UDim2.fromOffset((x1 + x2) / 2 * self.s, (y1 + y2) / 2 * self.s)
	f.Size = UDim2.fromOffset(math.max(len, 1), math.max(th * self.s, 1))
	f.Rotation = ang
	applyStyle(f, self)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(1, 0)
	c.Parent = f
	return f
end

-- Ломаная: линия через точки
function Pen:polyline(pts, th)
	for i = 1, #pts - 1 do
		self:line(pts[i][1], pts[i][2], pts[i + 1][1], pts[i + 1][2], th)
	end
end

-- Стрелка: линия + шеврон-наконечник
function Pen:arrow(x1, y1, x2, y2, th, headLen)
	self:line(x1, y1, x2, y2, th)
	local dx, dy = x2 - x1, y2 - y1
	local len = math.sqrt(dx * dx + dy * dy)
	if len < 0.01 then return end
	local ux, uy = dx / len, dy / len
	local hl = headLen or 4
	-- два «пера» наконечника под 32° к направлению
	local a1x = -ux * math.cos(math.rad(32)) - uy * math.sin(math.rad(32))
	local a1y = -uy * math.cos(math.rad(32)) + ux * math.sin(math.rad(32))
	local a2x = -ux * math.cos(math.rad(32)) + uy * math.sin(math.rad(32))
	local a2y = -uy * math.cos(math.rad(32)) - ux * math.sin(math.rad(32))
	self:line(x2, y2, x2 + a1x * hl, y2 + a1y * hl, th)
	self:line(x2, y2, x2 + a2x * hl, y2 + a2y * hl, th)
end

-- Точка на окружности: угол в градусах, 0° = вправо, по часовой (экран, y вниз)
local function onCircle(cx, cy, r, deg)
	local a = math.rad(deg)
	return cx + r * math.cos(a), cy + r * math.sin(a)
end

-- Дуга / кольцо: центр, диаметр, толщина, начальный угол, разворот
function Pen:ring(cx, cy, d, th, a0, sweep)
	local r = d * self.s / 2
	sweep = sweep or 360
	local n = U.clamp(math.ceil(math.abs(sweep) / 12), 3, 40)
	local step = sweep / n
	for i = 0, n - 1 do
		local a1 = a0 + step * i
		local a2 = a0 + step * (i + 1)
		local x1, y1 = onCircle(cx, cy, r / self.s, a1)
		local x2, y2 = onCircle(cx, cy, r / self.s, a2)
		-- удлиняем хорды, чтобы сегменты смыкались без «зубцов»
		local dx, dy = x2 - x1, y2 - y1
		local l = math.sqrt(dx * dx + dy * dy)
		if l > 0.001 then
			local ex = dx / l * (th * 0.35)
			local ey = dy / l * (th * 0.35)
			self:line(x1 - ex, y1 - ey, x2 + ex, y2 + ey, th)
		end
	end
end

-- Заливка полигона: построчный сканлайн (кадр-полоса на каждую пересечённую строку)
function Pen:poly(pts)
	-- границы
	local ymin, ymax = math.huge, -math.huge
	for _, p in ipairs(pts) do
		if p[2] < ymin then ymin = p[2] end
		if p[2] > ymax then ymax = p[2] end
	end
	local step = 1 -- шаг сетки в единицах дизайн-пространства
	local j = #pts
	for y = ymin + step * 0.5, ymax, step do
		local xs = {}
		for i = 1, #pts do
			local p1, p2 = pts[i], pts[j]
			if (p1[2] <= y and p2[2] > y) or (p2[2] <= y and p1[2] > y) then
				local t = (y - p1[2]) / (p2[2] - p1[2])
				xs[#xs + 1] = p1[1] + t * (p2[1] - p1[1])
			end
			j = i
		end
		table.sort(xs)
		for k = 1, #xs - 1, 2 do
			local w = xs[k + 1] - xs[k]
			if w > 0.05 then
				self:rect(xs[k], y - step * 0.5, w, step, 0)
			end
		end
	end
end

-- Сердце: параметрическая кривая, равномерно сэмплированная в полигон
function Pen:heart(cx, cy, scale)
	local pts = {}
	local n = 30
	for i = 0, n - 1 do
		local t = math.pi * 2 * i / n
		local x = 16 * math.sin(t) ^ 3
		local y = 13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t)
		pts[#pts + 1] = { cx + x * scale, cy - y * scale } -- y инвертирован (экран)
	end
	self:poly(pts)
end

-- Шестерёнка: полигон с зубьями + прорезка центра
function Pen:gear(cx, cy, rOuter, rInner, teeth, holeD)
	local pts = {}
	local seg = 360 / teeth
	for i = 0, teeth - 1 do
		local a = i * seg
		pts[#pts + 1] = { onCircle(cx, cy, rOuter, a + seg * 0.05) }
		pts[#pts + 1] = { onCircle(cx, cy, rOuter, a + seg * 0.38) }
		pts[#pts + 1] = { onCircle(cx, cy, rInner, a + seg * 0.50) }
		pts[#pts + 1] = { onCircle(cx, cy, rInner, a + seg * 0.88) }
	end
	self:poly(pts)
	if holeD and holeD > 0 then
		self:hole(cx, cy, holeD)
	end
end

-- Звезда (n лучей)
function Pen:star(cx, cy, rOuter, rInner, rays, rot)
	local pts = {}
	for i = 0, rays * 2 - 1 do
		local r = (i % 2 == 0) and rOuter or rInner
		pts[#pts + 1] = { onCircle(cx, cy, r, rot + 360 * i / (rays * 2)) }
	end
	self:poly(pts)
end

-- Волна (синусоида точками → polyline)
function Pen:wave(x0, x1, cy, amp, period, th)
	local pts = {}
	local n = 18
	for i = 0, n do
		local x = U.lerp(x0, x1, i / n)
		local y = cy + math.sin((x / period) * math.pi * 2) * amp
		pts[#pts + 1] = { x, y }
	end
	self:polyline(pts, th)
end

-- Сколько кадров создал пекарь (диагностика)
function NVG.cost(pen)
	return pen and pen.count or 0
end

-- Тангенциальный наконечник для дуг (стрелка по касательной)
local function arcHead(pen, cx, cy, r, deg, th, hl)
	local px, py = cx + r * math.cos(math.rad(deg)), cy + r * math.sin(math.rad(deg))
	local ux, uy = -math.sin(math.rad(deg)), math.cos(math.rad(deg)) -- касательная (по часовой)
	pen:arrow(px, py, px + ux * 1.1, py + uy * 1.1, th, hl or 4)
end

--────────────────── Публичный рендер ──────────────────
-- Рисует иконку name в host размером px. Возвращает pen (для диагностики).
-- opts: color, alpha, bg, grad = {c1, c2, rot}, state (второй аргумент художника)
function Icons.draw(host, px, name, opts)
	opts = opts or {}
	local fn = Icons.painters[name]
	local pen = NVG.pen(host, px, 24)
	pen:st({
		color = opts.color or Theme.colors.text,
		alpha = opts.alpha or 0,
		bg = opts.bg or Theme.colors.bg,
	})
	if opts.grad then
		pen:st({ grad = opts.grad })
	end
	if fn then
		local ok, err = pcall(fn, pen, opts.state)
		if not ok then
			warn("[NovaMusic] Иконка «" .. tostring(name) .. "»: " .. tostring(err))
		end
	end
	return pen
end

-- кастомные скроллбары, слайдеры, тултипы и тосты.

local TweenService = Nova.services.Tween

Nova.kit = Kit

--────────────────────────── Фабрика ──────────────────────────
function Kit.new(className, props, parent)
	local inst = Instance.new(className)
	props = props or {}
	for k, v in pairs(props) do
		if k ~= "Parent" then
			inst[k] = v
		end
	end
	inst.Parent = parent or props.Parent
	return inst
end

function Kit.corner(inst, r)
	local c = Instance.new("UICorner")
	if type(r) == "number" then
		c.CornerRadius = UDim.new(0, r)
	elseif r then
		c.CornerRadius = r
	else
		c.CornerRadius = UDim.new(1, 0)
	end
	c.Parent = inst
	return c
end

function Kit.stroke(inst, color, transparency, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color or Color3.fromRGB(255, 255, 255)
	s.Transparency = transparency or 0.88
	s.Thickness = thickness or 1
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = inst
	return s
end

function Kit.pad(inst, l, t, r, b)
	local p = Instance.new("UIPadding")
	p.PaddingLeft = UDim.new(0, l or 0)
	p.PaddingTop = UDim.new(0, t or 0)
	p.PaddingRight = UDim.new(0, r or 0)
	p.PaddingBottom = UDim.new(0, b or 0)
	p.Parent = inst
	return p
end

function Kit.grad(inst, c1, c2, rot, transparency)
	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(c1, c2)
	g.Rotation = rot or 90
	if transparency then
		g.Transparency = transparency
	end
	g.Parent = inst
	return g
end

function Kit.list(inst, direction, padding, sortOrder)
	local l = Instance.new("UIListLayout")
	l.FillDirection = direction or Enum.FillDirection.Vertical
	l.Padding = UDim.new(0, padding or 0)
	l.SortOrder = Enum.SortOrder.LayoutOrder
	if sortOrder then l.SortOrder = sortOrder end
	l.Parent = inst
	return l
end

function Kit.label(parent, props)
	local defaults = {
		BackgroundTransparency = 1,
		Font = Theme.fonts.semi,
		TextColor3 = Theme.colors.text,
		TextSize = 14,
		Text = "",
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}
	for k, v in pairs(props) do defaults[k] = v end
	return Kit.new("TextLabel", defaults, parent)
end

--────────────────────────── Твин-хелпер ──────────────────────────
function Kit.tween(inst, info, props)
	local ti = type(info) == "number" and Theme.motion.base or info
	local ok, tw = pcall(function() return TweenService:Create(inst, ti, props) end)
	if ok then tw:Play() return tw end
	return nil
end

--────────────────────────── Smart UI ──────────────────────────
-- Служебные элементы (скроллбары, подсказки) исчезают при бездействии
-- и возвращаются при наведении, вводе или смене трека.
local Smart = { items = {}, enabled = true, hoverOwners = {} }
Nova.smart = Smart

local function fadeTargetsFor(obj)
	local props = {}
	local cls = obj.ClassName
	if cls == "TextLabel" or cls == "TextButton" then
		props.TextTransparency = obj.TextTransparency
		props.BackgroundTransparency = obj.BackgroundTransparency
	elseif cls == "ImageLabel" or cls == "ImageButton" then
		props.ImageTransparency = obj.ImageTransparency
		props.BackgroundTransparency = obj.BackgroundTransparency
	else
		props.BackgroundTransparency = obj.BackgroundTransparency
	end
	-- прилагательный UIStroke гасим вместе
	local stroke = obj:FindFirstChildOfClass("UIStroke")
	if stroke then props._stroke = stroke end
	local pad = obj:FindFirstChildOfClass("UIPadding")
	return props, stroke
end

function Smart.register(obj, opts)
	opts = opts or {}
	local baseProps, stroke = fadeTargetsFor(obj)
	local item = {
		obj = obj,
		idle = opts.idle or 1.5,
		shown = true,
		token = 0,
		base = baseProps,
		stroke = stroke,
		strokeT = stroke and stroke.Transparency or 1,
	}
	Smart.items[#Smart.items + 1] = item
	if opts.owner then
		Smart.bindOwner(item, opts.owner)
	end
	-- события, которые «будят» элемент
	if opts.wake then
		for _, sig in ipairs(opts.wake) do
			U.trackConn(sig:Connect(function() Smart.pulse(item) end))
		end
	end
	return item
end

function Smart.bindOwner(item, owner)
	Smart.hoverOwners[owner] = Smart.hoverOwners[owner] or {}
	table.insert(Smart.hoverOwners[owner], item)
	U.trackConn(owner.MouseEnter:Connect(function() Smart.pulse(item) end))
end

function Smart.apply(item, shown)
	local obj = item.obj
	if shown then
		Kit.tween(obj, Theme.motion.base, item.base)
		if item.stroke then
			Kit.tween(item.stroke, Theme.motion.base, { Transparency = item.strokeT })
		end
	else
		local hideProps = {}
		for k in pairs(item.base) do
			if k ~= "_stroke" then hideProps[k] = 1 end
		end
		Kit.tween(obj, Theme.motion.slow, hideProps)
		if item.stroke then
			Kit.tween(item.stroke, Theme.motion.slow, { Transparency = 1 })
		end
	end
end

function Smart.pulse(item)
	if not Smart.enabled then return end
	item.token = item.token + 1
	local my = item.token
	if not item.shown then
		item.shown = true
		Smart.apply(item, true)
	end
	task.delay(item.idle, function()
		if item.token == my and Smart.enabled then
			item.shown = false
			Smart.apply(item, false)
		end
	end)
end

-- «Разбудить» всё, привязанное к владельцу (например, список при смене трека)
function Smart.pulseOwner(owner)
	local arr = Smart.hoverOwners[owner]
	if arr then
		for _, item in ipairs(arr) do Smart.pulse(item) end
	end
end

function Smart.setEnabled(on)
	Smart.enabled = on
	for _, item in ipairs(Smart.items) do
		item.token = item.token + 1
		item.shown = on
		Smart.apply(item, on)
	end
end

--────────────────────────── Кнопки ──────────────────────────
-- Иконка рисуется NVG внутрь держателя; перерисовка — только при смене состояния.
function Kit.iconButton(parent, opts)
	opts = opts or {}
	local btn = Kit.new("TextButton", {
		Name = "Btn_" .. tostring(opts.name or "btn"),
		AutoButtonColor = false,
		Text = "",
		BackgroundColor3 = opts.bg or Theme.colors.surface,
		BackgroundTransparency = opts.bgTrans or 0,
		Size = opts.size or UDim2.fromOffset(36, 36),
		Position = opts.pos or UDim2.new(),
		AnchorPoint = opts.anchor or Vector2.new(0, 0),
		Parent = parent,
	}, parent)
	Kit.corner(btn, opts.radius or Theme.radius.sm)
	local stroke
	if opts.stroke ~= false then
		stroke = Kit.stroke(btn, Theme.colors.white, 0.92, 1)
	end
	local holder = Kit.new("Frame", {
		Name = "IconHolder",
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromOffset(opts.iconSize or 18, opts.iconSize or 18),
		Parent = btn,
	})
	local scaler = Kit.new("UIScale", { Parent = holder })
	local state = { icon = opts.icon, opts = opts }

	local function drawIcon()
		for _, ch in ipairs(holder:GetChildren()) do
			if not ch:IsA("UIScale") then ch:Destroy() end
		end
		if state.icon then
			Icons.draw(holder, holder.Size.X.Offset, state.icon, {
				color = opts.iconColor or Theme.colors.text,
				alpha = opts.iconAlpha or 0,
				bg = opts.bg or Theme.colors.surface,
				grad = opts.iconGrad,
				state = state,
			})
		end
	end

	function btn.setIcon(name, iconOpts)
		state.icon = name
		if iconOpts then
			for k, v in pairs(iconOpts) do opts[k] = v end
		end
		drawIcon()
	end

	btn._holder = holder
	btn._scaler = scaler
	btn._opts = opts
	drawIcon()

	-- Ховер: подсветка + мягкий pop иконки
	local hoverBg = opts.hoverBg or Theme.colors.surfaceHi
	btn.MouseEnter:Connect(function()
		Kit.tween(btn, Theme.motion.fast, { BackgroundColor3 = hoverBg })
		Kit.tween(scaler, Theme.motion.pop, { Scale = opts.hoverScale or 1.08 })
		if opts.onHover then opts.onHover(true) end
	end)
	btn.MouseLeave:Connect(function()
		Kit.tween(btn, Theme.motion.base, { BackgroundColor3 = opts.bg or Theme.colors.surface })
		Kit.tween(scaler, Theme.motion.base, { Scale = 1 })
		if opts.onHover then opts.onHover(false) end
	end)
	btn.MouseButton1Down:Connect(function()
		Kit.tween(scaler, Theme.motion.instant, { Scale = (opts.hoverScale or 1.08) * 0.88 })
	end)
	btn.MouseButton1Up:Connect(function()
		Kit.tween(scaler, Theme.motion.spring, { Scale = opts.hoverScale or 1.08 })
	end)
	if opts.onClick then
		btn.Activated:Connect(function()
			opts.onClick(btn)
		end)
	end
	return btn
end

--────────────────────────── Кастомный скроллбар ──────────────────────────
-- Присоединяется к ScrollingFrame (ScrollBarThickness = 0), рисует свой бегунок,
-- поддерживает перетаскивание и регистрацию в Smart UI.
function Kit.scrollbar(scroll, parentTo, opts)
	opts = opts or {}
	local width = opts.width or 4
	local track = Kit.new("Frame", {
		Name = "ScrollTrack",
		BackgroundTransparency = 1,
		Size = UDim2.new(0, math.max(width + 10, 14), 1, 0),
		Position = UDim2.new(1, -(width + 10) - (opts.offset or 2), 0, 4),
		AnchorPoint = Vector2.new(0, 0),
		Parent = parentTo,
	}, parentTo)
	local thumb = Kit.new("Frame", {
		Name = "ScrollThumb",
		BackgroundColor3 = opts.color or Theme.colors.textFaint,
		BackgroundTransparency = 0.35,
		Size = UDim2.new(1, 0, 0, 40),
		Position = UDim2.new(0, 2, 0, 0),
		Parent = track,
	}, track)
	Kit.corner(thumb, math.floor(width / 2) + 1)

	local dragging = false
	local dragStartMouse, dragStartPos = 0, 0

	local function metrics()
		local view = scroll.AbsoluteWindowSize.Y
		local content = scroll.AbsoluteCanvasSize.Y
		local trackH = track.AbsoluteSize.Y - 8
		local maxScroll = math.max(content - view, 1)
		local ratio = U.clamp(view / math.max(content, 1), 0.08, 1)
		local thumbH = math.max(trackH * ratio, 30)
		return view, content, trackH, maxScroll, ratio, thumbH
	end

	local function sync()
		if not scroll.Parent then return end
		local view, content, trackH, maxScroll, ratio, thumbH = metrics()
		local canvasPos = scroll.CanvasPosition.Y
		thumb.Size = UDim2.new(1, -4, 0, thumbH)
		local t = U.clamp(canvasPos / maxScroll, 0, 1)
		thumb.Position = UDim2.new(0, 2, 0, 4 + t * (trackH - thumbH))
	end

	thumb.MouseEnter:Connect(sync)
	scroll:GetPropertyChangedSignal("CanvasPosition"):Connect(sync)
	U.trackConn(bus.LibChanged:Connect(function()
		task.defer(sync)
	end))
	track:GetPropertyChangedSignal("AbsoluteSize"):Connect(sync)
	task.defer(sync)

	-- Перетаскивание бегунка
	thumb.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStartMouse = UIS:GetMouseLocation().Y
			dragStartPos = scroll.CanvasPosition.Y
			local view, content, trackH, maxScroll, ratio, thumbH = metrics()
			local moveConn
			moveConn = UIS.InputChanged:Connect(function(m)
				if m.UserInputType == Enum.UserInputType.MouseMovement and dragging then
					local dy = UIS:GetMouseLocation().Y - dragStartMouse
					local k = maxScroll / math.max(trackH - thumbH, 1)
					scroll.CanvasPosition = Vector2.new(0, U.clamp(dragStartPos + dy * k, 0, maxScroll))
				end
			end)
			UIS.InputEnded:Connect(function(m)
				if m.UserInputType == Enum.UserInputType.MouseButton1 then
					dragging = false
					if moveConn then moveConn:Disconnect() end
				end
			end)
		end
	end)

	-- Клик по дорожке: прыжок
	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 and not dragging then
			local view, content, trackH, maxScroll, ratio, thumbH = metrics()
			local my = UIS:GetMouseLocation().Y
			local top = track.AbsolutePosition.Y
			local t = U.clamp((my - top - 4) / math.max(trackH, 1), 0, 1)
			Kit.tween(scroll, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
				CanvasPosition = Vector2.new(0, t * maxScroll)
			})
		end
	end)

	-- Smart UI: бегунок сам скрывается
	local item = Smart.register(thumb, { idle = 1.4, owner = opts.owner or scroll })
	-- широкая невидимая зона наведения у края
	local hit = Kit.new("Frame", {
		Name = "ScrollHit",
		BackgroundTransparency = 1,
		Size = UDim2.new(0, 16, 1, 0),
		Position = UDim2.new(1, -18, 0, 0),
		Parent = parentTo,
	}, parentTo)
	U.trackConn(hit.MouseEnter:Connect(function() Smart.pulse(item) end))
	Smart.pulse(item)
	task.delay(2.2, function()
		if item.shown then Smart.pulse(item) end -- через пару секунд сам уйдёт
		item.token = item.token + 1
		local my = item.token
		task.delay(item.idle, function()
			if item.token == my and Smart.enabled and not dragging then
				item.shown = false
				Smart.apply(item, false)
			end
		end)
	end)

	return { item = item, sync = sync, thumb = thumb, track = track }
end

--────────────────────────── Слайдер ──────────────────────────
-- value 0..1; поддерживает колесо мыши, перетаскивание, внешний sync.
function Kit.slider(parent, opts)

		Text = "",
		AutoButtonColor = false,
		BackgroundTransparency = 1,

		Parent = root,

	local track = Kit.new("Frame", {
		Name = "Track",
		BackgroundColor3 = opts.trackColor or Theme.colors.surfaceHi,

	Kit.corner(track, barH / 2)
	-- Заполнение: маска-контейнер с клипом
	local fillHolder = Kit.new("Frame", {
		Name = "FillHolder",
		BackgroundTransparency = 1,
		ClipsDescendants = true,
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 0, 0.5, 0),
		Size = UDim2.new(opts.value or 0, 0, 1, 0),
		Parent = track,
	}, track)
	local fill = Kit.new("Frame", {
		Name = "Fill",
		BackgroundColor3 = Theme.accent.flat,
		Size = UDim2.fromScale(1, 1),
		Parent = fillHolder,
	}, fillHolder)
	Kit.corner(fill, barH / 2)

	local knob = Kit.new("Frame", {
		Name = "Knob",
		BackgroundColor3 = Theme.colors.white,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(opts.value or 0, 0, 0.5, 0),
		Size = UDim2.fromOffset(0, 0),
		Parent = root,
	}, root)
	Kit.corner(knob, 999)
	Kit.stroke(knob, Theme.colors.white, 0.7, 1)
	local knobScaler = Kit.new("UIScale", { Parent = knob })

	local value = U.clamp(opts.value or 0, 0, 1)
	local dragging = false

	local function apply(v, animate)
		value = U.clamp(v, 0, 1)
		local info = animate and Theme.motion.fast or Theme.motion.instant
		Kit.tween(fillHolder, info, { Size = UDim2.new(value, 0, 1, 0) })
		Kit.tween(knob, info, { Position = UDim2.new(value, 0, 0.5, 0) })
	end

	local function fromX()
		local abs = hit.AbsoluteSize.X
		if abs <= 0 then return value end
		local mx = UIS:GetMouseLocation().X
		local rel = (mx - hit.AbsolutePosition.X) / abs
		return U.clamp(rel, 0, 1)
	end

	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			knob.Size = UDim2.fromOffset(13, 13)
			apply(fromX())
			if opts.onChange then opts.onChange(value, true) end
		end
	end)
	U.trackConn(UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 and dragging then
			dragging = false
			knob.Size = UDim2.fromOffset(0, 0)
			if opts.onChange then opts.onChange(value, false) end
		end
	end))
	U.trackConn(UIS.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			apply(fromX())
			if opts.onChange then opts.onChange(value, true) end
		end
	end))
	if opts.wheel ~= false then
		hit.MouseEnter:Connect(function()
			Kit.tween(knobScaler, Theme.motion.fast, { Scale = 1 })
			knob.Size = UDim2.fromOffset(11, 11)
		end)
		hit.MouseLeave:Connect(function()
			if not dragging then knob.Size = UDim2.fromOffset(0, 0) end
		end)
		hit.InputChanged:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseWheel then
				local step = opts.wheelStep or 0.05
				apply(value + (input.Position.Z > 0 and step or -step))
				if opts.onChange then opts.onChange(value, false) end
			end
		end)
	end

	apply(value)
	return {
		root = root,
		get = function() return value end,
		isDragging = function() return dragging end,
		set = function(v, animate) apply(v, animate) end,
		fill = fill,
	}
end

function Kit.bindTooltip(btn, text)
	btn.MouseEnter:Connect(function()
		if not tooltipState.label then return end
		tooltipState.token = tooltipState.token + 1
		local my = tooltipState.token
		task.delay(0.45, function()
			if tooltipState.token ~= my then return end
			local lbl = tooltipState.label
			lbl.Text = text
			-- ширина по тексту
			local ts = game:GetService("TextService")
			local ok, size = pcall(function()
				return ts:GetTextSize(text, 12, lbl.Font, Vector2.new(1000, 100))
			end)
			local w = (ok and size.X or 60) + 20
			lbl.Size = UDim2.fromOffset(w, 26)
			local abs = btn.AbsolutePosition
			lbl.Position = UDim2.fromOffset(abs.X + btn.AbsoluteSize.X / 2, abs.Y - 8)
			lbl.Visible = true
			lbl.TextTransparency = 1
			lbl.BackgroundTransparency = 1
			Kit.tween(lbl, Theme.motion.fast, { TextTransparency = 0, BackgroundTransparency = 0.06 })
		end)
	end)
	btn.MouseLeave:Connect(function()
		tooltipState.token = tooltipState.token + 1
		if tooltipState.label and tooltipState.label.Visible then
			local lbl = tooltipState.label
			Kit.tween(lbl, Theme.motion.fast, { TextTransparency = 1, BackgroundTransparency = 1 })
			task.delay(0.2, function()
				if lbl.TextTransparency >= 1 then lbl.Visible = false end
			end)
		end
	end)
end

-- Обрезка текста с «…» по фактической ширине
function Kit.fitText(label, maxPx)
	local ts = game:GetService("TextService")
	local ok, size = pcall(function()
		return ts:GetTextSize(label.Text, label.TextSize, label.Font, Vector2.new(10000, 1000))
	end)
	if not ok then return end
	if size.X <= maxPx then return end
	local t = label.Text
	while #t > 1 do
		t = t:sub(1, -2)
		local ok2, s2 = pcall(function()
			return ts:GetTextSize(t .. "…", label.TextSize, label.Font, Vector2.new(10000, 1000))
		end)
		if ok2 and s2.X <= maxPx then
			label.Text = t .. "…"
			return
		end
	end
	label.Text = "…"
end

-- Детерминированная обложка: один и тот же трек всегда выглядит одинаково
function Covers.make(parent, size, track)
	local seed = U.hash(track and track.key or "nova")
	local rng = U.seededRandom(seed)
	local pal = Covers.palettes[(seed % #Covers.palettes) + 1]
	local motif = seed % 3

	local card = Nova.kit.new("Frame", {
		Name = "Cover",
		BackgroundColor3 = pal.bg,
		BorderSizePixel = 0,
		Size = UDim2.fromOffset(size, size),
		ClipsDescendants = true,
		Parent = parent,
	}, parent)
	Nova.kit.corner(card, math.floor(size * 0.16))
	-- мягкая вертикальная глубина
	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(Color3.fromRGB(255, 255, 255), Color3.fromRGB(150, 150, 175))
	g.Rotation = 115
	g.Parent = card

	local pen = Nova.nvg.pen(card, size, 24)
	pen.s = size / 24
	pen.color = pal.a
	pen.alpha = 0

	if motif == 0 then
		-- «Орбиты»: смещённый градиентный диск + кольца + спутник
		local cx, cy = 9 + rng() * 4, 10 + rng() * 4
		pen.color = pal.a

		pen.color = pal.b
		pen.alpha = 0.55
		pen:ring(12, 12, 20, 1.4, 0, 360)
		pen:ring(12, 12, 25.5, 1.1, 40, 220)
		pen.alpha = 0
		pen:circle(19.4, 5.8, 3.4)
	elseif motif == 1 then
		-- «Волны»: три синусоиды с разной амплитудой
		pen.color = pal.a
		pen:wave(2, 22, 9 + rng() * 2, 3.4, 7 + rng() * 3, 1.7)
		pen.color = pal.b
		pen:wave(2, 22, 14 + rng() * 2, 2.4, 5 + rng() * 3, 1.4)
		pen.alpha = 0.5
		pen:wave(2, 22, 17.5, 1.6, 9, 1.1)
		pen.alpha = 0
		pen:circle(20, 5.6, 2.8)
	else
		-- «Пики»: треугольник, штрих-бары и точка

		pen:poly({ { 4, 18.5 }, { 11.5, 5.5 }, { 19, 18.5 } })

	return card

--────────────────────────── Аудио-движок ──────────────────────────
local Audio = {
	state = {
		track = nil, index = 0,
		playing = false, time = 0, duration = 0,
	},
	sound = nil,
	effects = {},
	simConn = nil,
	pollAcc = 0,
	broken = {},
}
Nova.audio = Audio

function Audio.applyVolume()
	if Audio.sound then
		Audio.sound.Volume = Nova.config.data.muted and 0 or Nova.config.data.volume
	end
	bus.VolumeChanged:Fire(Nova.config.data.volume, Nova.config.data.muted)
end

-- Цепочка эффектов: единый эквалайзер + отдельные инстансы
function Audio.buildEffects()
	local d = Nova.config.data.effects
	-- Эквалайзер: один инстанс, гейнсы складываются
	local low = d.bass and 8 or 0
	local mid = d.vocal and 5 or 0
	local high = d.treble and 5 or 0
	if Audio.eq then
		Audio.eq.Enabled = (low + mid + high) > 0
		Audio.eq.GainLow = low
		Audio.eq.GainMid = mid
		Audio.eq.GainHigh = high
	end
	if Audio.reverb then Audio.reverb.Enabled = d.reverb end
	if Audio.wide then Audio.wide.Enabled = d.wide end
	if Audio.pitch then Audio.pitch.Enabled = d.nightcore end
	if Audio.lofi then Audio.lofi.Enabled = d.lofi end
	bus.EffectChanged:Fire(d)
end

function Audio.mountEffects(sound)
	-- чистим старые
	for _, key in ipairs({ "eq", "reverb", "wide", "pitch", "lofi" }) do
		if Audio[key] then Audio[key]:Destroy() Audio[key] = nil end
	end
	Audio.eq = Instance.new("EqualizerSoundEffect")
	Audio.eq.Name = "NovaEQ"
	Audio.eq.Enabled = false
	Audio.eq.Parent = sound
	Audio.reverb = Instance.new("ReverbSoundEffect")
	Audio.reverb.DecayTime = 1.8
	Audio.reverb.WetLevel = 0.32
	Audio.reverb.DryLevel = 0.8
	Audio.reverb.Enabled = false
	Audio.reverb.Parent = sound
	Audio.wide = Instance.new("ChorusSoundEffect")
	Audio.wide.Depth = 0.55
	Audio.wide.Rate = 0.7
	Audio.wide.Mix = 0.35
	Audio.wide.Enabled = false
	Audio.wide.Parent = sound
	Audio.pitch = Instance.new("PitchShiftSoundEffect")
	Audio.pitch.Octave = 1.25 -- 1.0 у PitchShift — отсутствие сдвига
	Audio.pitch.Enabled = false
	Audio.pitch.Parent = sound
	Audio.lofi = Instance.new("DistortionSoundEffect")
	Audio.lofi.Level = 0.65
	Audio.lofi.Enabled = false
	Audio.lofi.Parent = sound
	Audio.buildEffects()
end

function Audio.setEffect(name, on)
	local d = Nova.config.data.effects
	if d[name] == nil then return end
	d[name] = on and true or false
	Nova.config.save()
	Audio.buildEffects()
end

local function stopSim()
	if Audio.simConn then
		Audio.simConn:Disconnect()
		Audio.simConn = nil
	end
end

local function startSim()
	stopSim()
	Audio.simConn = RunService.Heartbeat:Connect(function(dt)
		local st = Audio.state
		if not st.playing then return end
		st.time = st.time + dt * (Nova.config.data.speed or 1)
		if st.time >= st.duration then
			Audio.onTrackEnd()
		else
			Audio.pollAcc = Audio.pollAcc + dt
			if Audio.pollAcc >= 0.1 then
				Audio.pollAcc = 0
				bus.TimeUpdated:Fire(st.time, st.duration)
			end
		end
	end)
end

function Audio.onTrackEnd()
	local d = Nova.config.data
	local st = Audio.state
	if d.loop == "one" then
		Audio.seekTo(0)
		Audio.resume()
		return
	end
	Audio.next(false)
end

-- Полная загрузка трека в движок
function Audio.loadTrack(track, index, autoPlay)
	local st = Audio.state
	st.track = track
	st.index = index or track.index or 0
	st.duration = track.duration or 200
	st.time = 0
	stopSim()
	if Audio.sound then
		Audio.sound:Stop()
		Audio.sound:Destroy()
		Audio.sound = nil
	end
	if track.demo then
		-- Симуляция: демо-треки идут «в такт» без аудио ассета
		startSim()
		st.playing = autoPlay ~= false
	else
		local sound = Instance.new("Sound")
		sound.Name = "NovaTrack"
		sound.SoundId = track.id
		sound.Volume = Nova.config.data.muted and 0 or Nova.config.data.volume
		sound.PlaybackSpeed = Nova.config.data.speed or 1
		sound.Parent = SoundService
		Audio.sound = sound
		Audio.mountEffects(sound)
		Audio.applyVolume()
		-- предзагрузка + контроль ошибок
		task.spawn(function()
			local ok = pcall(function()
				ContentProvider:PreloadAsync({ sound })
			end)
			if not ok then return end
			if sound.IsLoaded and sound.TimeLength > 0 then
				st.duration = sound.TimeLength
				track.duration = sound.TimeLength
			end
		end)
		sound.Ended:Connect(function()
			if Audio.sound == sound then
				Audio.onTrackEnd()
			end
		end)
		st.playing = autoPlay ~= false
		if st.playing then
			sound:Play()
		end
	end
	Nova.config.data.lastTrack = track.key
	Nova.config.save()
	bus.TrackChanged:Fire(track, st.index)
	bus.StateChanged:Fire(st.playing)
	bus.TimeUpdated:Fire(0, st.duration)
end

function Audio.playTrack(track)
	local st = Audio.state
	if st.track and st.track.key == track.key then
		Audio.toggle()
		return
	end
	local idx = track.index or 1
	Audio.loadTrack(track, idx, true)
end

function Audio.resume()
	local st = Audio.state
	if not st.track then return end
	st.playing = true
	if st.track.demo then
		startSim()
	else
		if Audio.sound then
			-- после естественного конца трека надёжнее Play, чем Resume
			if st.time >= st.duration - 0.05 or st.time <= 0.01 then
				pcall(function() Audio.sound.TimePosition = math.max(st.time, 0) end)
				pcall(function() Audio.sound:Play() end)
			else
				pcall(function() Audio.sound:Resume() end)
			end
		end
	end
	bus.StateChanged:Fire(true)
end

function Audio.pause()
	local st = Audio.state
	st.playing = false
	if Audio.sound then Audio.sound:Pause() end
	-- симуляция просто замирает (simConn остаётся, dt не копится)
	bus.StateChanged:Fire(false)
end

function Audio.toggle()
	if Audio.state.playing then
		Audio.pause()
	else
		Audio.resume()
	end
end

function Audio.seekTo(t)
	local st = Audio.state
	st.time = U.clamp(t, 0, st.duration)
	if Audio.sound and st.track and not st.track.demo then
		pcall(function() Audio.sound.TimePosition = st.time end)
	end
	bus.TimeUpdated:Fire(st.time, st.duration)
end

function Audio.next(user)
	local st = Audio.state
	local tracks = Nova.library.tracks
	if #tracks == 0 then return end
	local idx = st.index
	if Nova.config.data.shuffle and #tracks > 1 then
		local pick = idx
		while pick == idx do
			pick = math.random(1, #tracks)
		end
		idx = pick
	else
		idx = idx + 1
		if idx > #tracks then
			if Nova.config.data.loop == "all" then
				idx = 1
			else
				idx = #tracks
				Audio.loadTrack(tracks[idx], idx, false)
				Audio.pause()
				return
			end
		end
	end
	Audio.loadTrack(tracks[idx], idx, true)
end

function Audio.prev()
	local st = Audio.state
	local tracks = Nova.library.tracks
	if #tracks == 0 then return end
	-- Если прошло больше 4 секунд — сначала «в начало»
	if st.time > 4 then
		Audio.seekTo(0)
		return
	end
	local idx = st.index - 1
	if idx < 1 then
		idx = (Nova.config.data.loop == "all") and #tracks or 1
	end
	Audio.loadTrack(tracks[idx], idx, true)
end

function Audio.setVolume(v)
	Nova.config.data.volume = U.clamp(v, 0, 1)
	if Nova.config.data.volume > 0 then Nova.config.data.muted = false end
	Nova.config.save()
	Audio.applyVolume()
end

function Audio.toggleMute()
	Nova.config.data.muted = not Nova.config.data.muted
	Nova.config.save()
	Audio.applyVolume()
end

function Audio.cycleLoop()
	local order = { off = "all", all = "one", one = "off" }
	Nova.config.data.loop = order[Nova.config.data.loop] or "off"
	Nova.config.save()
	bus.LoopChanged:Fire(Nova.config.data.loop)
end

function Audio.toggleShuffle()
	Nova.config.data.shuffle = not Nova.config.data.shuffle
	Nova.config.save()
	bus.ShuffleChanged:Fire(Nova.config.data.shuffle)
end

function Audio.setSpeed(s)
	Nova.config.data.speed = U.clamp(s, 0.5, 1.5)
	Nova.config.save()
	if Audio.sound then
		Audio.sound.PlaybackSpeed = Nova.config.data.speed
	end
end

function Audio.refreshLibrary()
	Nova.library.scan()
	if Audio.state.track then
		-- пересинхронизируем ссылку на текущий трек (мог обновиться)
		local fresh = Nova.library.findByKey(Audio.state.track.key)
		if fresh then
			Audio.state.track = fresh
			Audio.state.index = fresh.index or Audio.state.index
		end
	end
	bus.Toast:Fire("Библиотека обновлена: " .. tostring(#Nova.library.tracks) .. " трек(ов)", "ok")
end

-- Тик синхронизации времени для реальных звуков
U.trackConn(RunService.Heartbeat:Connect(function(dt)
	local st = Audio.state
	if st.playing and Audio.sound and st.track and not st.track.demo then
		Audio.pollAcc = Audio.pollAcc + dt
		if Audio.pollAcc >= 0.1 then
			Audio.pollAcc = 0
			local ok, pos = pcall(function() return Audio.sound.TimePosition end)
			if ok and pos then st.time = pos end
			local okL, len = pcall(function() return Audio.sound.TimeLength end)
			if okL and len and len > 0 then
				st.duration = len
				if st.track.duration ~= len then
					st.track.duration = len
				end
			end
			bus.TimeUpdated:Fire(st.time, st.duration)
		end

--══════════════════════════ Часть 8: Караоке-движок ══════════════════════════
-- Лирика берётся из JSON трека; если её нет — детерминированно генерируется
-- по сиду трека и bpm (демо-режим). Движок считает индекс строки и слова
-- и рассылает события bus.KaraokeLine / bus.KaraokeWord для UI.

local U = Nova.util
local bus = Nova.bus

local Lyrics = {}
Nova.lyrics = Lyrics

--────────────────────────── Генератор демо-лирики ──────────────────────────
local BANK_ADJ = { "неоновый", "полночный", "тихий", "стеклянный", "зимний", "электрический", "бархатный", "хрустальный", "ветреный", "синий", "последний", "новый", "бесконечный", "прохладный", "далёкий" }
local BANK_NOUN = { "рассвет", "город", "пульс", "дождь", "маяк", "ветер", "силуэт", "горизонт", "трамвай", "эфир", "кадр", "маршрут", "прибой", "океан", "фонарь", "сигнал", "пейзаж", "закат", "отпечаток", "полёт" }
local BANK_VERB = { "плывёт", "звенит", "дышит", "молчит", "исчезает", "находит", "помнит", "ждёт", "летит", "горит", "догоняет", "растворяется", "остаётся", "просыпается" }
local BANK_VERB_WE = { "плывём", "летим", "молчим", "ждём", "горим", "дышим", "идём", "исчезаем", "просыпаемся", "помним" }
local BANK_CONN = { "и", "в", "на", "сквозь", "за", "между", "без", "около", "перед" }

local PATTERNS = {
	function(r, bank) return bank.adj .. " " .. bank.noun .. " " .. bank.verb end,
	function(r, bank) return "Мы " .. bank.verbWe .. " " .. bank.conn .. " " .. bank.adj .. " " .. bank.noun end,
	function(r, bank) return bank.noun .. " " .. bank.verb .. " " .. bank.conn .. " " .. bank.noun2 end,
	function(r, bank) return "Это " .. bank.adj .. " " .. bank.noun .. ", " .. bank.conn .. " " .. bank.noun2 end,
	function(r, bank) return "Не " .. bank.verbWe .. " — " .. bank.noun .. " " .. bank.verb end,
	function(r, bank) return bank.conn .. " " .. bank.adj .. " " .. bank.noun .. " " .. bank.verbWe .. " свет" end,
	function(r, bank) return bank.noun .. ", " .. bank.noun2 .. " и " .. bank.noun3 end,
	function(r, bank) return "Только " .. bank.adj .. " " .. bank.noun .. " " .. bank.verb end,
}

-- Первая буква заглавная только для ASCII (безопасно для UTF-8)
local function capFirst(s)
	local b = s:byte(1)
	if b and b >= 97 and b <= 122 then
		return string.char(b - 32) .. s:sub(2)
	end
	return s
end

-- Строит таблицу банка слов из генератора
local function makeBank(rng)
	local function pickInt(arr) return math.floor(rng(1, #arr + 0.999)) end
	return {
		adj = BANK_ADJ[pickInt(BANK_ADJ)],
		noun = BANK_NOUN[pickInt(BANK_NOUN)],
		noun2 = BANK_NOUN[pickInt(BANK_NOUN)],
		noun3 = BANK_NOUN[pickInt(BANK_NOUN)],
		verb = BANK_VERB[pickInt(BANK_VERB)],
		verbWe = BANK_VERB_WE[pickInt(BANK_VERB_WE)],
		conn = BANK_CONN[pickInt(BANK_CONN)],
	}
end

-- Публичный провайдер: массив строк {t, dur, text, words}
function Lyrics.get(track)
	if track.lyrics and #track.lyrics > 0 then
		local out = {}
		for i, l in ipairs(track.lyrics) do
			local nextT = track.lyrics[i + 1] and track.lyrics[i + 1].t or (l.t + 6)
			local words = {}
			for w in l.line:gmatch("%S+") do
				words[#words + 1] = w
			end
			out[#out + 1] = {
				t = l.t,
				dur = math.max(nextT - l.t, 0.8),
				text = l.line,
				words = words,
			}
		end
		return out
	end
	-- Процедурная лирика для демо: стабильна для конкретного трека
	local seed = U.hash(track.key or track.title)
	local rng = U.seededRandom(seed)
	local beat = 60 / (track.bpm or 104)
	local lineDur = math.max(beat * 8, 2.2) -- 2 такта
	local count = math.max(math.floor((track.duration or 200) / lineDur), 6)
	local out = {}
	for i = 1, count do
		local bank = makeBank(rng)
		local patIdx = math.floor(rng(1, #PATTERNS + 0.999))
		local text = PATTERNS[patIdx](rng, bank)
		text = capFirst(text:sub(1, 1)) .. text:sub(2)
		local words = {}
		for w in text:gmatch("%S+") do
			words[#words + 1] = w
		end
		local t = (i - 1) * lineDur
		-- Каждая 4-я строка — «инструментальный» перерыв
		if i % 4 == 0 then
			out[#out + 1] = { t = t, dur = lineDur, text = "· · ·", words = { "·", "·", "·" }, instrumental = true }
		else
			out[#out + 1] = { t = t, dur = lineDur, text = text, words = words }
		end
	end
	return out
end

-- Набор случайных слов для эффекта «вылетающих слов» при смене трека
function Lyrics.burstWords(track, n)
	local words = {}
	if track and track.lyrics then
		for _, l in ipairs(track.lyrics) do
			for w in l.line:gmatch("%S+") do
				local clean = w:gsub("[%p]", "")
				if #clean > 2 then words[#words + 1] = clean end
			end
		end
	end
	if #words < n then
		for _ = 1, n do
			local r = U.hash(tostring(math.random()) .. tostring(os.clock()))
			words[#words + 1] = BANK_NOUN[r % #BANK_NOUN + 1]
		end
	end
	local out = {}
	local used = {}
	local guard = 0
	while #out < math.min(n, #words) and guard < 60 do
		guard = guard + 1
		local idx = math.random(1, #words)
		if not used[idx] then
			used[idx] = true
			out[#out + 1] = words[idx]
		end
	end
	return out
end

local Kara = {
	timeline = {},
	lineIndex = 0,
	wordIndex = 0,
	enabled = true,
	track = nil,
}
Nova.karaoke = Kara

function Kara.setTrack(track)
	Kara.track = track
	Kara.lineIndex = 0
	Kara.wordIndex = 0
	Kara.timeline = track and Lyrics.get(track) or {}
	bus.KaraokeLine:Fire(nil, 0)
end

-- Вес слова: длина + пробел (длинные слова «поются» дольше)
local function lineWeights(line)
	local weights, total = {}, 0
	for i, w in ipairs(line.words) do
		local wt = U.clamp(#w, 1, 14) + 1
		weights[i] = wt
		total = total + wt
	end
	return weights, total
end

function Kara.update(time)
	if not Kara.enabled or #Kara.timeline == 0 then return end
	-- поиск строки (с продвижением кэша)
	local idx = U.clamp(Kara.lineIndex, 0, #Kara.timeline)
	while idx < #Kara.timeline and Kara.timeline[idx + 1].t <= time do
		idx = idx + 1
	end
	-- откат, если время отмотали назад
	while idx > 0 and Kara.timeline[idx].t > time do
		idx = idx - 1
	end
	if idx ~= Kara.lineIndex then
		Kara.lineIndex = idx
		Kara.wordIndex = 0
		if idx == 0 then
			bus.KaraokeLine:Fire(nil, 0)
		else
			bus.KaraokeLine:Fire(Kara.timeline[idx], idx)
		end
	end
	-- индекс слова внутри строки: слово активно с момента своего НАЧАЛА
	local line = Kara.timeline[idx]
	if line and not line.instrumental and time >= line.t then
		local weights, total = lineWeights(line)
		local frac = U.clamp((time - line.t) / math.max(line.dur, 0.1), 0, 1)
		local acc = 0
		local wi = 0
		for i, w in ipairs(weights) do
			if frac >= acc - 0.0001 then
				wi = i
				acc = acc + w / total
			else
				break
			end
		end
		if wi ~= Kara.wordIndex then
			Kara.wordIndex = wi
			bus.KaraokeWord:Fire(wi)
		end
	end
end

function Kara.currentLine()
	return Kara.timeline[Kara.lineIndex], Kara.wordIndex
end

local function mountScreen()
	local screen
	local candidates = {}
	local okHui, hui = pcall(function()
		if type(gethui) == "function" then return gethui() end
		return nil
	end)
	if okHui and hui then candidates[#candidates + 1] = hui end
	candidates[#candidates + 1] = game:GetService("CoreGui")
	local pg = Nova.services.Player and Nova.services.Player:FindFirstChildOfClass("PlayerGui")
	if pg then candidates[#candidates + 1] = pg end
	candidates[#candidates + 1] = game

	for _, holder in ipairs(candidates) do
		local ok = pcall(function()
			local s = Instance.new("ScreenGui")
			s.Name = "NovaMusicUI"
			s.ResetOnSpawn = false
			s.IgnoreGuiInset = true
			s.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
			s.DisplayOrder = 9999
			s.Parent = holder
			screen = s
		end)
		if ok and screen then break end
	end
	if not screen then
		error("[NovaMusic] Не удалось смонтировать ScreenGui")
	end
	UI.screen = screen
	U.trackInstance(screen)
	return screen
end

-- перетаскивание: общий хелпер
function Kit.makeDraggable(handle, target)
	local dragging = false
	local startMouse, startPos
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			startMouse = UIS:GetMouseLocation()
			startPos = Vector2.new(target.Position.X.Offset, target.Position.Y.Offset)
		end
	end)
	U.trackConn(UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 and dragging then
			dragging = false
			Nova.config.data.pos = { target.Position.X.Offset, target.Position.Y.Offset }
			Nova.config.save()
		end
	end))
	U.trackConn(UIS.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local m = UIS:GetMouseLocation()
			target.Position = UDim2.new(0.5, startPos.X + (m.X - startMouse.X), 0.5, startPos.Y + (m.Y - startMouse.Y))
		end
	end))
end

function UI.isRowActive(track)
	return Audio.state.track ~= nil and Audio.state.track.key == track.key
end

function UI.updateCounts()
	UI.countLabel.Text = tostring(#Nova.library.tracks) .. " треков"
	local n = 0
	for _ in ipairs(Nova.config.data.liked) do n = n + 1 end
	UI.likedLabel.Text = tostring(n) .. " в избранном"
end

function UI.setView(v)
	UI.view = v
	local function markActive(navBtn, active)
		Kit.tween(navBtn.btn, Theme.motion.base, { BackgroundTransparency = active and 0.25 or 1 })
		Kit.tween(navBtn.bar, Theme.motion.base, { BackgroundTransparency = active and 0 or 1 })
		Kit.tween(navBtn.label, Theme.motion.base, { TextColor3 = active and Theme.colors.text or Theme.colors.textDim })
	end
	markActive(UI.navAll, v == "all")
	markActive(UI.navLiked, v == "liked")
	UI.viewTitle.Text = (v == "all") and "Твоя музыка" or "Избранное"
	UI.rebuildList()
end

function UI.setQuery(q)
	UI.query = U.lowerRu(q)
	UI.rebuildList()
end

function UI.filteredTracks()
	local out = {}
	for _, t in ipairs(Nova.library.tracks) do
		local okView = (UI.view == "all") or (UI.view == "liked" and Nova.library.isLiked(t))
		local okQuery = #UI.query == 0
			or U.lowerRu(t.title):find(UI.query, 1, true)
			or U.lowerRu(t.artist):find(UI.query, 1, true)
		if okView and okQuery then
			out[#out + 1] = t
		end
	end
	return out
end

function UI.rebuildList()
	-- полная очистка: строки и пустые заглушки
	for _, ch in ipairs(UI.list:GetChildren()) do
		if (ch:IsA("TextButton") and ch.Name:sub(1, 4) == "Row_") or ch.Name == "Empty" then
			ch:Destroy()
		end
	end
	UI.rows = {}
	local tracks = UI.filteredTracks()
	for i, t in ipairs(tracks) do
		UI.rows[i] = buildRow(t, i)
	end
	if #tracks == 0 then
		Kit.label(UI.list, {
			Name = "Empty",
			Size = UDim2.new(1, -10, 0, 120),
			LayoutOrder = 9999,
			Text = "Ничего не найдено\nДобавьте JSON в NovaMusicData/Songs",
			Font = Theme.fonts.semi,
			TextSize = 14,
			TextColor3 = Theme.colors.textFaint,
			TextXAlignment = Enum.TextXAlignment.Center,
			TextYAlignment = Enum.TextYAlignment.Center,
		})
	end
	UI.updateCounts()
	UI.viewSub.Text = tostring(#tracks) .. " трек(ов) · обновлено " .. os.date("%H:%M")
	UI.updateRowStates()
	task.defer(function()
		if UI.scroll and UI.scroll.sync then UI.scroll.sync() end
	end)
end

function UI.updateRowStates()
	for _, r in ipairs(UI.rows) do
		local active = UI.isRowActive(r.track)
		local playing = active and Audio.state.playing
		r.title.TextColor3 = active and Theme.accent.flat or Theme.colors.text
		Kit.tween(r.frame, Theme.motion.base, { BackgroundTransparency = active and 0 or 1, BackgroundColor3 = active and Theme.colors.rowActive or Theme.colors.row })
		r.eqHolder.Visible = playing
		if not (playing) then
			Kit.fadeIcon(r.playHolder, 1)
			r.indexLbl.TextTransparency = 0
		else
			r.indexLbl.TextTransparency = 1
			Kit.fadeIcon(r.playHolder, 1)
		end
	end
end

-- Эквалайзер активной строки: живые столбики в такт
function UI.startEQTicker()
	if UI.eqConn then return end
	local acc = 0
	UI.eqConn = RunService.Heartbeat:Connect(function(dt)
		acc = acc + dt
		local st = Audio.state
		if not (st.playing and st.track) then return end
		local t = os.clock()
		local row = nil
		for _, r in ipairs(UI.rows) do
			if r.track and UI.isRowActive(r.track) then row = r break end
		end
		if row and row.eqHolder.Visible then
			local bpmF = (st.track.bpm or 104) / 104
			for b = 1, 3 do
				local h = 5 + math.abs(math.sin(t * 6 * bpmF + b * 1.7)) * 11
				row.eqBars[b].Size = UDim2.fromOffset(3, h)
			end
		end
	end)
	U.trackConn(UI.eqConn)
end

local function karaokeBuildLine(line)
	for _, ch in ipairs(UI.karaokeLineHolder:GetChildren()) do
		if ch:IsA("TextLabel") then ch:Destroy() end
	end
	UI.karaokeWordLabels = {}
	UI.karaokeLineText = line
	if not line then return end
	if line.instrumental then
		local l = Kit.label(UI.karaokeLineHolder, {
			Text = "· · ·",
			Font = F.bold,
			TextSize = 18,
			TextColor3 = P.textFaint,
			TextTransparency = 1,
		})
		Kit.tween(l, Theme.motion.smooth, { TextTransparency = 0.25 })
		UI.karaokeWordLabels[1] = l
		return
	end
	local scale = 1
	local maxLen = 34
	if #line.text > maxLen then scale = math.max(maxLen / #line.text, 0.6) end
	local size = math.floor(21 * scale)
	for wi, w in ipairs(line.words) do
		local l = Kit.label(UI.karaokeLineHolder, {
			Text = w,
			Font = F.black,
			TextSize = size,
			TextColor3 = P.textFaint,
			LayoutOrder = wi,
			AutomaticSize = Enum.AutomaticSize.XY,
			Size = UDim2.fromOffset(0, size + 6),
		})
		UI.karaokeWordLabels[wi] = l
	end
	-- вход строки: лёгкий подъём
	for _, l in ipairs(UI.karaokeWordLabels) do
		l.TextTransparency = 1
	end
	task.delay(0.01, function()
		for i, l in ipairs(UI.karaokeWordLabels) do
			task.delay(0.03 * i, function()
				if l.Parent then
					Kit.tween(l, Theme.motion.base, { TextTransparency = 0 })
				end
			end)
		end
	end)
end

local function karaokeHighlight(wi)
	local labels = UI.karaokeWordLabels
	if not labels then return end
	for i, l in ipairs(labels) do
		if l.Text == "· · ·" then return end
		if i < wi then
			Kit.tween(l, Theme.motion.fast, { TextColor3 = P.text })
		elseif i == wi then
			Kit.tween(l, Theme.motion.pop, { TextColor3 = Theme.accent.flat })
		else
			Kit.tween(l, Theme.motion.fast, { TextColor3 = P.textFaint })
		end
	end
end

-- публикация локальных обработчиков для части 11
UI.karaokeBuildLine = karaokeBuildLine
UI.karaokeHighlight = karaokeHighlight

function UI.wordBurst(track)
	if not UI.karaokeOn then return end
	local layer = UI.burstLayer
	if not layer then return end
	local words = Lyrics.burstWords(track, 8)
	local origin = UI.root.AbsolutePosition
	for i, w in ipairs(words) do
		local l = Kit.label(layer, {
			Text = w,
			Font = (i % 2 == 0) and F.black or F.bold,
			TextSize = 14 + math.random(0, 14),
			TextColor3 = (i % 3 == 0) and Theme.accent.flat or P.white,
			TextTransparency = 0.35,
			Rotation = math.random(-16, 16),
			Position = UDim2.fromOffset(
				origin.X + Theme.layers.sidebarW + math.random(30, 600),
				origin.Y + 90 + math.random(0, 330)
			),
			ZIndex = 51,
		})
		Kit.tween(l, TweenInfo.new(1.3 + math.random() * 0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.In), {
			Position = l.Position - UDim2.fromOffset(0, 90 + math.random(0, 70)),
			TextTransparency = 1,
			Rotation = l.Rotation + math.random(-8, 8),
		})
		task.delay(2.3, function()
			if l.Parent then l:Destroy() end
		end)
	end
end

function UI.toggleSettings()
	local p = UI.settingsPanel
	if not p then return end
	local isOpen = p:GetAttribute("Open") == true
	if isOpen then
		p:SetAttribute("Open", false)
		Kit.tween(p, Theme.motion.base, { Position = UDim2.new(1, 12, 0, 12) })
		task.delay(0.3, function()
			if p:GetAttribute("Open") == false then p.Visible = false end
		end)
	else
		p:SetAttribute("Open", true)
		p.Visible = true
		Kit.tween(p, Theme.motion.spring, { Position = UDim2.new(1, -12, 0, 12) })
	end
	-- подсветка кнопки эффектов (isOpen — значение ДО переключения)
	UI.fxBtn.setIcon("sliders", { iconColor = isOpen and P.textDim or Theme.accent.flat })
end

function UI.toggleHelp()
	local m = UI.helpModal
	if not m then return end
	local open = m.Visible
	if open then
		Kit.tween(m, Theme.motion.fast, { BackgroundTransparency = 1 })
		task.delay(0.18, function() m.Visible = false end)
	else
		m.Visible = true
		m.BackgroundTransparency = 1
		Kit.tween(m, Theme.motion.spring, { BackgroundTransparency = 0.06 })
	end
end

function UI.setKaraoke(on)
	UI.karaokeOn = on
	Nova.config.data.karaoke = on
	Nova.config.save()

	if strip then
		strip.Visible = true

			Kit.tween(strip, Theme.motion.smooth, { Position = UDim2.new(0, Theme.layers.sidebarW, 1, -(Theme.layers.playerH + 96)) })

			Kit.tween(strip, Theme.motion.smooth, { Position = UDim2.new(0, Theme.layers.sidebarW, 1, -Theme.layers.playerH) })
			Kit.tween(UI.list, Theme.motion.smooth, { Size = UDim2.new(1, -Theme.layers.sidebarW, 1, -(Theme.layers.headerH + Theme.layers.playerH)) })

	end

--────────────────────────── Синхронизация UI с движками ──────────────────────────
local function wireUI()
	-- Смена трека
	U.trackConn(bus.TrackChanged:Connect(function(track, index)
		if not track then return end
		-- обложка «сейчас играет»
		for _, ch in ipairs(UI.nowCover:GetChildren()) do ch:Destroy() end
		Covers.make(UI.nowCover, 60, track)
		UI.nowTitle.Text = track.title
		Kit.fitText(UI.nowTitle, 148)
		UI.nowArtist.Text = track.artist .. (track.demo and "  ·  демо" or "")
		local liked = Nova.library.isLiked(track)
		UI.heartNow.setIcon(liked and "heartFill" or "heartOutline", { iconColor = liked and Theme.accent.flat or P.textDim })
		UI.timeCur.Text = "0:00"
		UI.timeDur.Text = U.fmtTime(Audio.state.duration)
		UI.seek.set(0)
		UI.updateRowStates()
		UI.autoScrollTo(track)
		Nova.smart.pulseOwner(UI.list)
		UI.wordBurst(track)
		if UI.karaokeBuildLine then UI.karaokeBuildLine(nil) end
		UI.refreshTransportIcons()
	end))

	-- Плей / пауза
	U.trackConn(bus.StateChanged:Connect(function(playing)
		UI.btnPlay.setIcon(playing and "pause" or "play", { iconColor = P.black })
		local sc = UI.btnPlay._scaler
		if sc then Kit.tween(sc, Theme.motion.spring, { Scale = 1 }) end
		UI.updateRowStates()
	end))

	-- Время
	U.trackConn(bus.TimeUpdated:Connect(function(t, dur)
		if UI.seek and not UI.seek.isDragging() then
			UI.seek.set(dur > 0 and t / dur or 0)
		end
		UI.timeCur.Text = U.fmtTime(t)
		UI.timeDur.Text = U.fmtTime(dur)
	end))

	-- Громкость
	U.trackConn(bus.VolumeChanged:Connect(function(v, muted)
		if UI.vol and not UI.vol.isDragging() then
			UI.vol.set(muted and 0 or v)
		end
		UI.volIcon.setIcon(muted and "volumeMute" or "volume", { iconColor = P.text })
	end))

	-- Караоке: строка и слово
	U.trackConn(bus.KaraokeLine:Connect(function(line)
		if UI.karaokeBuildLine then UI.karaokeBuildLine(line) end
	end))
	U.trackConn(bus.KaraokeWord:Connect(function(wi)
		if UI.karaokeHighlight then UI.karaokeHighlight(wi) end
	end))

	-- Библиотека
	U.trackConn(bus.LibChanged:Connect(function()
		UI.rebuildList()
		UI.updateCounts()
	end))

	-- Акцент
	U.trackConn(bus.AccentChanged:Connect(function(key)
		UI.applyAccent()
		for k, dot in pairs(UI.accentDots or {}) do
			local st = dot:FindFirstChildOfClass("UIStroke")
			if st then st.Transparency = (k == key) and 0.1 or 0.6 end
		end
		UI.syncEffectChips()
		for k, chip in pairs(UI.speedChips or {}) do
			if math.abs((Nova.config.data.speed or 1) - k) < 0.01 then
				chip.BackgroundColor3 = Theme.accent.flat
			end
		end
	end))
end

-- Прокрутка списка к активному треку
function UI.autoScrollTo(track)
	for i, r in ipairs(UI.rows) do
		if r.track and r.track.key == track.key then
			local target = (i - 1) * (UI.ROW_H + UI.ROW_GAP)
			local view = UI.list.AbsoluteWindowSize.Y
			local center = math.max(target - (view - UI.ROW_H) / 2, 0)
			Kit.tween(UI.list, TweenInfo.new(0.4, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				CanvasPosition = Vector2.new(0, center),
			})
			return
		end
	end
end

--────────────────────────── Горячие клавиши ──────────────────────────
function Nova.setupHotkeys()
	U.trackConn(UIS.InputBegan:Connect(function(input, gameProcessed)
		if gameProcessed then return end
		if not Nova.running then return end
		local kc = input.KeyCode
		local tracks = UI.filteredTracks()

		if kc == Enum.KeyCode.Up or kc == Enum.KeyCode.Down then
			if #tracks == 0 then return end
			local cur = 0
			if Audio.state.track then
				for i, t in ipairs(tracks) do
					if t.key == Audio.state.track.key then cur = i break end
				end
			end
			local nxt
			if kc == Enum.KeyCode.Up then
				nxt = (cur <= 1) and 1 or cur - 1
			else
				nxt = (cur >= #tracks) and #tracks or cur + 1
			end
			if tracks[nxt] then
				Audio.loadTrack(tracks[nxt], tracks[nxt].index or nxt, true)
			end
		elseif kc == Enum.KeyCode.Space then
			Audio.toggle()
		elseif kc == Enum.KeyCode.Left then
			Audio.seekTo(Audio.state.time - 5)
		elseif kc == Enum.KeyCode.Right then
			Audio.seekTo(Audio.state.time + 5)
		elseif kc == Enum.KeyCode.M then
			Audio.toggleMute()
		elseif kc == Enum.KeyCode.L then
			Audio.cycleLoop()
		elseif kc == Enum.KeyCode.S then
			Audio.toggleShuffle()
		elseif kc == Enum.KeyCode.K then
			UI.setKaraoke(not UI.karaokeOn)
		elseif kc == Enum.KeyCode.F then
			if UI.searchBox then
				UI.root.Visible = true
				UI.searchBox:CaptureFocus()
			end
		elseif kc == Enum.KeyCode.Escape then
			if UI.helpModal and UI.helpModal.Visible then UI.toggleHelp() end
			local sp = UI.settingsPanel
			if sp and sp:GetAttribute("Open") then UI.toggleSettings() end
			if UI.searchBox then UI.searchBox:ReleaseFocus() end
		elseif kc == Enum.KeyCode.RightControl then
			local r = UI.root
			r.Visible = not r.Visible
			if r.Visible then
				local sc = r:FindFirstChildOfClass("UIScale")
				if not sc then
					sc = Instance.new("UIScale")
					sc.Parent = r
				end
				sc.Scale = 0.96
				Kit.tween(sc, Theme.motion.spring, { Scale = 1 })
			else
				bus.Toast:Fire("Right Ctrl — вернуть плеер", "info")
			end
		end
	end))
end

--────────────────────────── Инициализация ──────────────────────────
function Nova.init()
	if not Nova.running then return end
	Nova.running = true
	math.randomseed(math.floor(os.clock() * 100000) % 2147483647)

	-- 1) Файлы, конфиг, библиотека
	Nova.library.init()

	-- 2) Акцент из конфига
	if Theme.accents[Nova.config.data.accent] then
		Theme.setAccent(Nova.config.data.accent)
	end

	-- 3) Интерфейс
	UI.buildAll()
	wireUI()
	UI.setView("all")
	UI.karaokeOn = Nova.config.data.karaoke
	UI.setKaraoke(Nova.config.data.karaoke)
	UI.refreshTransportIcons()
	UI.applyAccent()
	UI.syncEffectChips()
	UI.startEQTicker()
	Nova.smart.setEnabled(Nova.config.data.smartUI)
	Audio.applyVolume()
	Nova.setupHotkeys()

	-- 4) Последний трек (без автозапуска)
	local okLast = false
	if type(Nova.config.data.lastTrack) == "string" then
		local t = Nova.library.findByKey(Nova.config.data.lastTrack)
		if t then
			Audio.loadTrack(t, t.index or 1, false)
			okLast = true
		end
	end

	-- 5) Интро и появление окна
	UI.root.Visible = false
	local rootScale = Instance.new("UIScale")
	rootScale.Scale = 0.94
	rootScale.Parent = UI.root
	local function reveal()
		UI.root.Visible = true
		Kit.tween(rootScale, Theme.motion.spring, { Scale = 1 })
	end
	UI.runIntro(reveal)

	task.delay(2.3, function()
		if Nova.running then
			bus.Toast:Fire("Nova Music готова · " .. tostring(#Nova.library.tracks) .. " трек(ов)" .. (okLast and "" or " · демо-режим"), "ok")
		end
	end)

	Nova.exportAPI()
end

--────────────────────────── Выгрузка ──────────────────────────
function Nova.unload()
	if not Nova.running then return end
	Nova.running = false
	-- остановить звук
	if Audio.sound then
		pcall(function() Audio.sound:Stop() end)
		pcall(function() Audio.sound:Destroy() end)
		Audio.sound = nil
	end
	-- симуляция
	if Audio.simConn then
		Audio.simConn:Disconnect()
		Audio.simConn = nil
	end
	-- соединения
	for _, c in ipairs(Nova.conns) do
		pcall(function() c:Disconnect() end)
	end
	Nova.conns = {}
	-- интерфейс
	if UI.screen then
		pcall(function() UI.screen:Destroy() end)
		UI.screen = nil
	end
	if getgenv then
		getgenv().NovaMusic = nil
	end
	print("[NovaMusic] Выгружен. Спасибо, что слушали.")
end

--────────────────────────── Публичный API ──────────────────────────
function Nova.exportAPI()
	if not (getgenv and type(getgenv) == "function") then return end
	getgenv().NovaMusic = {
		version = Nova.version,
		Toggle = function()
			if UI.root then
				UI.root.Visible = not UI.root.Visible
			end
		end,
		Show = function()
			if UI.root then UI.root.Visible = true end
		end,
		Hide = function()
			if UI.root then UI.root.Visible = false end
		end,
		Unload = Nova.unload,
		API = {
			play = function(indexOrKey)
				local t
				if type(indexOrKey) == "number" then
					t = Nova.library.tracks[indexOrKey]
				else
					t = Nova.library.findByKey(tostring(indexOrKey))
				end
				if t then Audio.loadTrack(t, t.index or 1, true) end
				return t ~= nil
			end,
			next = function() Audio.next(true) end,
			prev = function() Audio.prev() end,
			pause = function() Audio.pause() end,
			resume = function() Audio.resume() end,
			toggle = function() Audio.toggle() end,
			seek = function(sec) Audio.seekTo(tonumber(sec) or 0) end,
			setVolume = function(v) Audio.setVolume(tonumber(v) or 0.8) end,
			toggleMute = function() Audio.toggleMute() end,
			setEffect = function(name, on) Audio.setEffect(name, on) end,
			setSpeed = function(s) Audio.setSpeed(tonumber(s) or 1) end,
			cycleLoop = function() Audio.cycleLoop() end,
			toggleShuffle = function() Audio.toggleShuffle() end,
			setKaraoke = function(on) UI.setKaraoke(on and true or false) end,
			setAccent = function(key)
				if Theme.accents[key] then
					Theme.setAccent(key)
					Nova.config.data.accent = key
					Nova.config.save()
				end
			end,
			refresh = function() Audio.refreshLibrary() end,
			getTracks = function()
				local out = {}
				for i, t in ipairs(Nova.library.tracks) do
					out[i] = { title = t.title, artist = t.artist, demo = t.demo, duration = t.duration }
				end
				return out
			end,
			getState = function()
				local st = Audio.state
				return {
					playing = st.playing, time = st.time, duration = st.duration,
					track = st.track and st.track.title or nil,
					volume = Nova.config.data.volume, loop = Nova.config.data.loop,
					shuffle = Nova.config.data.shuffle,
				}
			end,
		},
	}
end

-- Точка входа: запуск под защитой — любая ошибка будет ВИДНА в консоли,
-- а не «молчаливо» ломать загрузку скрипта
local okInit, errInit = pcall(Nova.init)
if not okInit then
	local msg = "[NovaMusic] Ошибка запуска: " .. tostring(errInit)
	print(msg)
	pcall(function() warn(msg) end)
	if getgenv and type(getgenv) == "function" then
		pcall(function() getgenv().NovaMusic = nil end)
	end
end
