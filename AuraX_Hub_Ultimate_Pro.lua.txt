-- // CLEANUP REGISTRY (powers the complete Unload) -------------------
local Registry = {
    connections = {}, -- RBXScriptConnections to disconnect
    objects     = {}, -- Instances to destroy
    tasks       = {}, -- custom stop/cleanup callbacks
}

local function TrackConnection(conn)
    if conn then table.insert(Registry.connections, conn) end
    return conn
end

local function TrackObject(inst)
    if inst then table.insert(Registry.objects, inst) end
    return inst
end

local function TrackTask(fn)
    if fn then table.insert(Registry.tasks, fn) end
end

-- Safe instance factory: per-property pcall so one bad prop on an
-- exotic executor can never break the whole build. Track = false for
-- short-lived ephemeral objects (rings, bursts, toasts).
local function New(className, props, parent, track)
    local inst = Instance.new(className)
    if props then
        for k, v in pairs(props) do
            local ok = pcall(function() inst[k] = v end)
            if not ok then
                warn("[AuraX] skipped unsupported property: " .. className .. "." .. tostring(k))
            end
        end
    end
    if parent then
        local ok = pcall(function() inst.Parent = parent end)
        if not ok then
            warn("[AuraX] could not parent: " .. className)
            pcall(function() inst:Destroy() end)
            return nil
        end
    end
    if track ~= false then TrackObject(inst) end
    return inst
end

-- // SMALL UTILITIES -------------------------------------------------
local TI = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function Tween(inst, props, ti)
    if not inst then return nil end
    local ok, tw = pcall(function()
        return TweenService:Create(inst, ti or TI, props)
    end)
    if ok and tw then
        tw:Play()
        return tw
    end
    return nil
end

-- // CENTRAL FRAME UPDATER (single connection, error-capped) ---------
local Updaters    = {}
local FailCounts  = {}

local function AddUpdater(fn)
    table.insert(Updaters, fn)
    FailCounts[#Updaters] = 0
end

TrackConnection(RunService.RenderStepped:Connect(function(dt)
    local t = os.clock()
    for i = 1, #Updaters do
        local fn = Updaters[i]
        if fn then
            local ok, err = pcall(fn, dt, t)
            if not ok then
                FailCounts[i] = (FailCounts[i] or 0) + 1
                if FailCounts[i] == 1 then
                    warn("[AuraX] updater error (suppressed): " .. tostring(err))
                end
                if FailCounts[i] >= 10 then
                    Updaters[i] = function() end -- auto-disable, never spam console
                    pcall(function() Notify("A module errored out and was disabled", "warn") end)
                end
            end
        end
    end
end))

-- // THEME SYSTEM ------------------------------------------------------
local Themes = {
    ["Neon Purple"] = Color3.fromRGB(168, 85, 247),
    ["Cyan"]        = Color3.fromRGB(56, 214, 250),
    ["Emerald"]     = Color3.fromRGB(64, 220, 160),
    ["Sunset"]      = Color3.fromRGB(255, 128, 110),
    ["Custom"]      = Color3.fromRGB(120, 200, 255),
}

local function Soften(c)
    return Color3.new(c.R * 0.62, c.G * 0.62, c.B * 0.62)
end

local ThemeState = {
    Accent = Themes["Neon Purple"],
    Soft   = Soften(Themes["Neon Purple"]),
}

local ThemedItems = {} -- {obj, kind}: stroke|fill|text|grad|part|scroll|viewport
local ThemeHooks  = {} -- extra callbacks on theme change (preset rings, etc.)

local function MarkTheme(inst, kind)
    if inst then table.insert(ThemedItems, {obj = inst, kind = kind}) end
end

local function AddThemeHook(fn)
    if fn then table.insert(ThemeHooks, fn) end
end

local function ApplyTheme(name)
    local accent
    if name == "Custom" then
        accent = Color3.fromRGB(
            math.clamp(Settings.CustomR, 0, 255),
            math.clamp(Settings.CustomG, 0, 255),
            math.clamp(Settings.CustomB, 0, 255)
        )
    else
        accent = Themes[name] or Themes["Neon Purple"]
    end
    Settings.Theme    = name
    ThemeState.Accent = accent
    ThemeState.Soft   = Soften(accent)

    for i = 1, #ThemedItems do
        local item = ThemedItems[i]
        local obj  = item.obj
        if obj and obj.Parent then
            pcall(function()
                if item.kind == "stroke" then
                    Tween(obj, {Color = accent})
                elseif item.kind == "fill" then
                    Tween(obj, {BackgroundColor3 = accent})
                elseif item.kind == "text" then
                    Tween(obj, {TextColor3 = accent})
                elseif item.kind == "grad" then
                    obj.Color = ColorSequence.new(accent, ThemeState.Soft)
                elseif item.kind == "part" then
                    Tween(obj, {Color = accent})
                elseif item.kind == "scroll" then
                    obj.ScrollBarImageColor3 = accent
                elseif item.kind == "viewport" then
                    obj.Ambient    = Color3.new(0.14 + accent.R * 0.20, 0.14 + accent.G * 0.20, 0.17 + accent.B * 0.20)
                    obj.LightColor = accent:Lerp(Color3.new(1, 1, 1), 0.55)
                end
            end)
        end
    end

    for i = 1, #ThemeHooks do
        pcall(ThemeHooks[i], name, accent)
    end
end

-- // CHARACTER TRACKING ------------------------------------------------
-- NOTE: we never touch Motor6D.Transform on the live character. All
-- skeletal/motor work is avoided by design (no "motor transform" errors).
local Char = { character = nil, humanoid = nil, root = nil }

local CharacterHooks = {} -- fn(ch) re-apply character-bound FX on respawn
local ScriptReady    = false

local function AddCharHook(fn)
    if fn then table.insert(CharacterHooks, fn) end
end

local function RunCharacterHooks(ch)
    for i = 1, #CharacterHooks do
        local ok, err = pcall(CharacterHooks[i], ch)
        if not ok then
            warn("[AuraX] character hook error (suppressed): " .. tostring(err))
        end
    end
end

HandleCharacter = function(ch)
    if not ch then return end
    Char.character = ch
    Char.humanoid  = nil
    Char.root      = nil
    task.spawn(function()
        local okH, h = pcall(function()
            return ch:FindFirstChildOfClass("Humanoid") or ch:WaitForChild("Humanoid", 8)
        end)
        local okR, r = pcall(function()
            return ch:WaitForChild("HumanoidRootPart", 8)
        end)
        Char.humanoid = (okH and h) or nil
        Char.root     = (okR and r) or nil
        if ScriptReady then
            RunCharacterHooks(ch)
        end
    end)
    if ScriptReady then
        RunCharacterHooks(ch)
    end
end

TrackConnection(LocalPlayer.CharacterAdded:Connect(HandleCharacter))
TrackConnection(LocalPlayer.CharacterRemoving:Connect(function()
    Char.character = nil
    Char.humanoid  = nil
    Char.root      = nil
end))

-- // GUI PARENT RESOLUTION --------------------------------------------
local function ResolveGuiParent()
    local ok, hui = pcall(function()
        if type(gethui) == "function" then return gethui() end
        return nil
    end)
    if ok and hui then return hui end

    local CoreGui = game:GetService("CoreGui")
    local canCore = pcall(function()
        local probe = Instance.new("Folder")
        probe.Name = "_aura_probe"
        probe.Parent = CoreGui
        probe:Destroy()
    end)
    if canCore then return CoreGui end

    return LocalPlayer.PlayerGui
end

local GuiParent = ResolveGuiParent()

local function ParentGui(gui)
    if type(syn) == "table" and type(syn.protect_gui) == "function" then
        pcall(syn.protect_gui, gui)
    end
    local ok = pcall(function() gui.Parent = GuiParent end)
    if not ok then
        pcall(function() gui.Parent = LocalPlayer.PlayerGui end)
    end
end

BuildPreview = function()
    pcall(function()
        if not PreviewViewport then return end
        if PreviewModel then
            pcall(function() PreviewModel:Destroy() end)
            PreviewModel = nil
        end
        local ch = LocalPlayer.Character
        if not (ch and ch.Parent) then
            if PreviewName then PreviewName.Text = "waiting for character..." end
            return
        end
        ch.Archivable = true
        local clone = ch:Clone()
        if not clone then
            if PreviewName then PreviewName.Text = "clone failed" end
            return
        end
        for _, d in ipairs(clone:GetDescendants()) do
            if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript") or d:IsA("Sound") or d:IsA("Highlight") then
                d:Destroy()
            elseif d:IsA("BasePart") then
                pcall(function() d.Anchored = true end)
            end
        end
        local hum = clone:FindFirstChildOfClass("Humanoid")
        if hum then
            pcall(function() hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end)
        end
        -- R6 rigs rest in a T-pose; fold the arms down into a calm stance
        if clone:FindFirstChild("Torso") then
            local rs = clone:FindFirstChild("Right Shoulder")
            local ls = clone:FindFirstChild("Left Shoulder")
            if rs and rs:IsA("Motor6D") then
                pcall(function() rs.Transform = CFrame.Angles(math.rad(92), 0, 0) end)
            end
            if ls and ls:IsA("Motor6D") then
                pcall(function() ls.Transform = CFrame.Angles(math.rad(92), 0, 0) end)
            end
        end
        local size = clone:GetExtentsSize()
        PreviewHeight = math.max(size.Y, 2)
        clone:PivotTo(CFrame.new(0, PreviewHeight / 2 + 0.05, 0))
        clone.Parent = PreviewWorld
        PreviewModel = clone
        pcall(function()
            PreviewCam.CFrame = CFrame.lookAt(
                Vector3.new(0.4, PreviewHeight * 0.64, -PreviewHeight * 2.15),
                Vector3.new(0, PreviewHeight * 0.5, 0)
            )
        end)
        if PreviewName then PreviewName.Text = LocalPlayer.Name end
    end)
end

end)

local function MakeDraggable(handle, target, opts)
    opts = opts or {}
    local dragging, moved = false, false
    local startInput, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            startInput = Vector2.new(input.Position.X, input.Position.Y)
            startPos = Vector2.new(target.Position.X.Offset, target.Position.Y.Offset)
            if CloseAllDropdowns then pcall(CloseAllDropdowns) end
        end
    end)
    TrackConnection(UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local dx = input.Position.X - startInput.X
            local dy = input.Position.Y - startInput.Y
            if math.abs(dx) + math.abs(dy) > 6 then moved = true end
            if moved then
                local nx, ny = startPos.X + dx, startPos.Y + dy
                if opts.clamp then
                    local cam = GetCam()
                    if cam then
                        local vps = cam.ViewportSize
                        local sz = target.AbsoluteSize
                        nx = math.clamp(nx, 8, math.max(8, vps.X - sz.X - 8))
                        ny = math.clamp(ny, 8, math.max(8, vps.Y - sz.Y - 8))
                    end
                end
                target.Position = UDim2.fromOffset(nx, ny)
            end
        end
    end))
    TrackConnection(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                if opts.onRelease then pcall(opts.onRelease, moved) end
            end
        end
    end))
end

MakeDraggable(TitleBar, Window, {clamp = true})
MakeDraggable(FloatBtn, FloatBtn, {
    clamp = true,
    onRelease = function(moved)
        if not moved then pcall(ToggleMenu) end
    end,
})

local function AttachSliderDrag(pad, onMove)
    local dragging = false
    pad.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            if CloseAllDropdowns then pcall(CloseAllDropdowns) end
            onMove(input)
        end
    end)
    TrackConnection(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            onMove(input)
        end
    end))
    TrackConnection(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))
end

-- Builds an orthonormal CFrame whose X axis (cylinder axis) lies along
-- `normal` — used to lay rings flat on slopes. (Fixed raycast alignment.)
local function GroundBasis(pos, normal)
    local x = normal
    if x.Magnitude < 0.05 then x = Vector3.new(0, 1, 0) end
    x = x.Unit
    local ref = (math.abs(x.Y) > 0.92) and Vector3.new(1, 0, 0) or Vector3.new(0, 1, 0)
    local y = (ref - x * ref:Dot(x))
    if y.Magnitude < 0.01 then y = Vector3.new(0, 0, 1) end
    y = y.Unit
    local z = x:Cross(y)
    return CFrame.fromMatrix(pos, x, y, z)
end

local function GroundRaycast(origin, dist)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Char.character, FXFolder}
    return Workspace:Raycast(origin, Vector3.new(0, -dist, 0), params)
end

            New("Part", {

local CharConns = {}
local lastJumpTrigger = 0

local function DisconnectCharConns()
    for i = 1, #CharConns do
        pcall(function() CharConns[i]:Disconnect() end)
    end
    CharConns = {}
end

local function SpawnJumpRing()
    pcall(function()
        local root = Char.root
        if not (root and root.Parent) then return end
        local origin = root.Position
        local hit = GroundRaycast(origin, 160)
        local pos, normal
        if hit then
            pos = hit.Position
            normal = hit.Normal
        else
            pos = origin - Vector3.new(0, root.Size.Y * 0.5 + 2.4, 0)
            normal = Vector3.new(0, 1, 0)
        end
        local basis = GroundBasis(pos + normal * 0.08, normal)

        local ring = New("Part", {
            Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.34, 3.4, 3.4),
            CFrame = basis, Anchored = true, CanCollide = false, CanQuery = false,
            CanTouch = false, Material = Enum.Material.Neon,

        }, FXFolder, false)

            Color = ThemeState.Accent, Range = 12, Brightness = 1.6,
        }, ring, false)
        Tween(ring, {
            Size = Vector3.new(0.04, 10.4, 10.4), Transparency = 1,
        }, TweenInfo.new(0.55, Enum.EasingStyle.Quart, Enum.EasingDirection.Out))
        Tween(light, {Brightness = 0, Range = 16}, TweenInfo.new(0.55))

        task.delay(0.08, function()
            pcall(function()
                local r2 = New("Part", {
                    Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.26, 2.2, 2.2),
                    CFrame = basis, Anchored = true, CanCollide = false, CanQuery = false,
                    CanTouch = false, Material = Enum.Material.Neon,
                    Color = ThemeState.Accent, Transparency = 0.3, CastShadow = false,
                }, FXFolder, false)
                Tween(r2, {
                    Size = Vector3.new(0.03, 7, 7), Transparency = 1,
                }, TweenInfo.new(0.5, Enum.EasingStyle.Quart, Enum.EasingDirection.Out))
                task.delay(0.55, function() pcall(function() r2:Destroy() end) end)
            end)
        end)

        local burst = New("Part", {
            Size = Vector3.new(0.4, 0.4, 0.4), Transparency = 1,
            CFrame = CFrame.new(pos + normal * 0.3), Anchored = true,
            CanCollide = false, CanQuery = false, CanTouch = false,
        }, FXFolder, false)
        local em = New("ParticleEmitter", {
            Color = ColorSequence.new(ThemeState.Accent),
            LightEmission = 1, LightInfluence = 0, Brightness = 1.5,
            Size = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.22),
                NumberSequenceKeypoint.new(1, 0),
            }),
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.1),
                NumberSequenceKeypoint.new(1, 1),
            }),
            Lifetime = NumberRange.new(0.35, 0.6),
            Speed = NumberRange.new(5, 10),
            SpreadAngle = Vector2.new(38, 38),
            Acceleration = Vector3.new(0, -26, 0),
            Rate = 0,
        }, burst, false)
        em:Emit(14)

        task.delay(0.65, function() pcall(function() ring:Destroy() end) end)
        task.delay(1.1, function() pcall(function() burst:Destroy() end) end)
    end)
end

local function JumpTrigger()
    if not Settings.JumpCircle then return end
    if os.clock() - lastJumpTrigger < 0.4 then return end
    lastJumpTrigger = os.clock()
    SpawnJumpRing()
end

        end

end)

        SpawnSparkBurst(hit.Position, math.floor(7 + 7 * math.clamp(Settings.ClickBurst, 0.5, 3)))
        local basis = GroundBasis(hit.Position + hit.Normal * 0.05, hit.Normal)

            Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.12, 1.2, 1.2),

local HistMax, HistHead = 64, 0

AddUpdater(function(dt, t)
    if not Settings.Cursor then return end
    local m = Vector2.new(200, 200)
    pcall(function() m = UserInputService:GetMouseLocation() end)
    CursorRoot.Position = UDim2.fromOffset(m.X, m.Y)
    HistHead = (HistHead % HistMax) + 1
    Hist[HistHead] = m

    local col
    if Settings.CursorColor == "Rainbow" then
        col = Color3.fromHSV((t * 0.4) % 1, 0.72, 1)
    elseif Settings.CursorColor == "White" then
        col = Color3.fromRGB(240, 240, 255)
    elseif Settings.CursorColor == "Ice" then
        col = Color3.fromRGB(150, 220, 255)

        col = Color3.fromRGB(255, 120, 170)
    else
        col = ThemeState.Accent
    end
    curDot.BackgroundColor3 = col
    curRingStroke.Color = col
    curCross1.BackgroundColor3 = col
    curCross2.BackgroundColor3 = col

    if Settings.CursorTrail then
        for i = 1, #Ghosts do
            local back = i * 3
            local idx = ((HistHead - back - 1) % HistMax) + 1
            local p = Hist[idx] or m
            local g = Ghosts[i]
            g.Position = UDim2.fromOffset(p.X - m.X + 1, p.Y - m.Y + 1)
            g.BackgroundColor3 = col
            g.Visible = true

local AspectRatios = {
    ["21:9"] = 21 / 9,
    ["16:9"] = 16 / 9,
    ["4:3"]  = 4 / 3,
    ["1:1"]  = 1,
}

local function ApplyBars(animate)
    pcall(function()
        local cam = GetCam()
        if not cam then return end
        local vp = cam.ViewportSize
        local top, bottom, left, right = 0, 0, 0, 0
        if Settings.Letterbox then
            local h = vp.Y * (Settings.LetterboxHeight / 100)
            top, bottom = h, h
        elseif Settings.AspectMode ~= "Off" then
            local r = AspectRatios[Settings.AspectMode]
            if r then
                local cur = vp.X / math.max(vp.Y, 1)
                if cur > r then
                    local w = (vp.X - vp.Y * r) / 2
                    left, right = w, w
                else
                    local h = (vp.Y - vp.X / r) / 2
                    top, bottom = h, h
                end
            end
        end
        local ti = animate and TweenInfo.new(0.4, Enum.EasingStyle.Sine, Enum.EasingDirection.Out) or TI
        Tween(Bars.top,    {Size = UDim2.new(0, 0, 0, top)}, ti)
        Tween(Bars.bottom, {Size = UDim2.new(0, 0, 0, bottom)}, ti)
        Tween(Bars.left,   {Size = UDim2.new(0, left, 0, 0)}, ti)
        Tween(Bars.right,  {Size = UDim2.new(0, right, 0, 0)}, ti)
        local vh = (top > 1)
        TopHairline.Visible = vh
        BottomHairline.Visible = vh
    end)
end

local function SpawnBolt()
    pcall(function()
        local base = nil
        local root = Char.root
        local cam = GetCam()
        if root and root.Parent then
            base = root.Position
        elseif cam then
            base = cam.CFrame.Position
        end
        if not base then return end
        local ang = math.rad(RNG:NextNumber(0, 360))
        local dist = RNG:NextNumber(24, 70)
        local tx = base.X + math.cos(ang) * dist
        local tz = base.Z + math.sin(ang) * dist
        local top = Vector3.new(tx + RNG:NextNumber(-26, 26), 150, tz + RNG:NextNumber(-26, 26))
        local bottom = Vector3.new(tx, 2, tz)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {Char.character, FXFolder}
        local hit = Workspace:Raycast(top, Vector3.new(0, -400, 0), params)
        if hit then bottom = hit.Position end

        local pts = {}
        local SEGS = 7
        for i = 0, SEGS do
            local a = i / SEGS
            local fade = (1 - a) * 3.4
            pts[#pts + 1] = Vector3.new(
                Lerp(top.X, bottom.X, a) + RNG:NextNumber(-fade, fade),
                Lerp(top.Y, bottom.Y, a),
                Lerp(top.Z, bottom.Z, a) + RNG:NextNumber(-fade, fade)
            )
        end
        pts[#pts] = bottom

        for i = 1, #pts - 1 do
            local a, b = pts[i], pts[i + 1]
            local len = (b - a).Magnitude
            if len > 0.1 then
                local seg = New("Part", {
                    Size = Vector3.new(0.26, 0.26, len),
                    CFrame = CFrame.lookAt((a + b) / 2, b),
                    Anchored = true, CanCollide = false, CanQuery = false,
                    CanTouch = false, Material = Enum.Material.Neon,
                    Color = Color3.fromRGB(214, 226, 255), Transparency = 0.05,
                    CastShadow = false,
                }, FXFolder, false)
                New("PointLight", {
                    Color = Color3.fromRGB(200, 216, 255), Range = 14, Brightness = 2,
                }, seg, false)
                task.delay(0.14, function() pcall(function() seg:Destroy() end) end)
            end
        end
        LightningFlash(0.5)
        task.delay(0.13, function() LightningFlash(0.3) end)
    end)
end

SetLightning = function(on)
    Settings.Lightning = on
    if on and not LightningActive then
        LightningActive = true
        task.spawn(function()
            while LightningActive do
                local waitT = RNG:NextNumber(4, 11)
                local t0 = os.clock()
                while LightningActive and os.clock() - t0 < waitT do
                    task.wait(0.25)
                end
                if LightningActive then
                    SpawnBolt()
                end
            end
        end)
    elseif not on then
        LightningActive = false
    end
end

ApplySkyPreset = function(name)
    Settings.SkyPreset = name
    pcall(function()
        if OurSky then pcall(function() OurSky:Destroy() end); OurSky = nil end
        if OurAtmo then pcall(function() OurAtmo:Destroy() end); OurAtmo = nil end
        if name == "Roblox Default" then
            for k, v in pairs(OriginalLighting) do
                pcall(function() Lighting[k] = v end)
            end
            if OriginalAtmosphere and OriginalAtmoProps then
                for k, v in pairs(OriginalAtmoProps) do
                    pcall(function() OriginalAtmosphere[k] = v end)
                end
            end
            if OriginalSky and OriginalSkyProps then
                for k, v in pairs(OriginalSkyProps) do
                    pcall(function() OriginalSky[k] = v end)
                end
            end
            return
        end
        local p = SkyPresets[name]
        if not p then return end
        Lighting.ClockTime      = p.ClockTime
        Lighting.Brightness     = p.Brightness
        Lighting.Ambient        = p.Ambient
        Lighting.OutdoorAmbient = p.Outdoor
        Lighting.FogEnd         = p.FogEnd
        Lighting.FogStart       = p.FogEnd * 0.08
        if p.Void then
            Lighting.FogColor = ThemeState.Accent:Lerp(Color3.new(0, 0, 0), 0.55)
        else
            Lighting.FogColor = p.FogColor
        end
        local atmo = OriginalAtmosphere
        if not atmo then
            if not (OurAtmo and OurAtmo.Parent) then
                OurAtmo = New("Atmosphere", {Name = "AuraX_Atmo"}, Lighting)
            end
            atmo = OurAtmo
        end
        for k, v in pairs(p.Atmo) do
            pcall(function() atmo[k] = v end)
        end
        if OriginalSky then
            pcall(function() OriginalSky.StarCount = p.Stars end)
            pcall(function() OriginalSky.SunAngularSize = p.Sun end)
            pcall(function() OriginalSky.MoonAngularSize = p.Moon end)
        else
            if not (OurSky and OurSky.Parent) then
                OurSky = New("Sky", {Name = "AuraX_Sky"}, Lighting)
            end
            pcall(function() OurSky.StarCount = p.Stars end)
            pcall(function() OurSky.SunAngularSize = p.Sun end)
            pcall(function() OurSky.MoonAngularSize = p.Moon end)
            pcall(function()
                OurSky.CelestialBodiesShown = (p.Stars > 0 or p.Sun > 0 or p.Moon > 0)
            end)
        end
    end)
end

-- // COMPLETE UNLOAD --------------------------------------------------------
local Unloaded = false

Unload = function()
    if Unloaded then return end
    Unloaded = true

    -- 1) stop effect loops, restore character + world toggles
    for i = 1, #Registry.tasks do
        pcall(Registry.tasks[i])
    end

    -- 2) disconnect every tracked connection
    for i = 1, #Registry.connections do
        pcall(function() Registry.connections[i]:Disconnect() end)
    end

    -- 3) restore Lighting exactly as we found it
    pcall(function()
        for k, v in pairs(OriginalLighting) do
            pcall(function() Lighting[k] = v end)
        end
        if OriginalAtmosphere and OriginalAtmoProps then
            for k, v in pairs(OriginalAtmoProps) do
                pcall(function() OriginalAtmosphere[k] = v end)
            end
        end
        if OriginalSky and OriginalSkyProps then
            for k, v in pairs(OriginalSkyProps) do
                pcall(function() OriginalSky[k] = v end)
            end
        end
    end)

    -- 4) restore camera + input state
    pcall(function()
        local cam = GetCam()
        if cam then cam.FieldOfView = OriginalFOV end
    end)
    pcall(function() UserInputService.MouseIconEnabled = true end)

    -- 5) destroy every tracked instance (GUIs, FX folder, blur, CC, sky...)
    for i = 1, #Registry.objects do
        pcall(function() Registry.objects[i]:Destroy() end)
    end

    _G.__AURAX_PRO_UNLOAD = nil
    _G.__AURAX_UNLOADING  = nil
    print("[AuraX] Ultimate Pro unloaded - all client effects removed.")
end

_G.__AURAX_PRO_UNLOAD = Unload
