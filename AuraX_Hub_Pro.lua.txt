-- // UTILITIES ------------------------------------------------------
local function new(className, props)
    local inst = Instance.new(className)
    local parent = nil
    if props then
        for key, value in pairs(props) do
            if key == "Parent" then
                parent = value
            else
                inst[key] = value
            end
        end
    end
    inst.Parent = parent
    return inst
end

local function corner(inst, radius)
    return new("UICorner", { CornerRadius = UDim.new(0, radius or 10), Parent = inst })
end

local function stroke(inst, color, thickness, transparency)
    return new("UIStroke", {
        Color = color or Theme.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0.92,
        Parent = inst,
    })
end

local function gradient(inst, c1, c2, rotation)
    return new("UIGradient", {
        Color = ColorSequence.new(c1, c2),
        Rotation = rotation or 0,
        Parent = inst,
    })
end

local function pad(inst, left, right, top, bottom)
    return new("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
        Parent = inst,
    })
end

local function tween(inst, seconds, props, style, direction)
    local info = TweenInfo.new(
        seconds,
        style or Enum.EasingStyle.Quad,
        direction or Enum.EasingDirection.Out
    )
    local t = TweenService:Create(inst, info, props)
    t:Play()
    return t
end

local function rainbowSequence()
    local keypoints = {}
    for i = 0, 6 do
        table.insert(keypoints, ColorSequenceKeypoint.new(i / 6, Color3.fromHSV(i / 6, 1, 1)))
    end
    return ColorSequence.new(keypoints)
end

local function lerpColor(a, b, alpha)
    return Color3.new(
        a.R + (b.R - a.R) * alpha,
        a.G + (b.G - a.G) * alpha,
        a.B + (b.B - a.B) * alpha
    )
end

-- // LIGHTING SNAPSHOT ----------------------------------------------
local LIGHT_PROPS = {
    "Brightness", "Ambient", "OutdoorAmbient", "ClockTime",
    "FogColor", "FogStart", "FogEnd", "GlobalShadows",
    "ExposureCompensation", "EnvironmentDiffuseScale", "EnvironmentSpecularScale",
}

local savedAtmo, savedAtmoExisted
local savedSky, savedSkyExisted

local function snapshotLighting()
    local snap = {}
    for _, key in ipairs(LIGHT_PROPS) do
        snap[key] = Lighting[key]
    end
    AuraX.lightSaved = snap

    local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
    savedAtmoExisted = atmo ~= nil
    if atmo then
        savedAtmo = {
            Density = atmo.Density, Offset = atmo.Offset, Color = atmo.Color,
            Decay = atmo.Decay, Glare = atmo.Glare, Haze = atmo.Haze,
        }
    end

    local sky = Lighting:FindFirstChildOfClass("Sky")
    savedSkyExisted = sky ~= nil
    if sky then
        savedSky = {
            SkyboxBk = sky.SkyboxBk, SkyboxDn = sky.SkyboxDn, SkyboxFt = sky.SkyboxFt,
            SkyboxLf = sky.SkyboxLf, SkyboxRt = sky.SkyboxRt, SkyboxUp = sky.SkyboxUp,
            StarCount = sky.StarCount, SunAngularSize = sky.SunAngularSize,
            MoonAngularSize = sky.MoonAngularSize, CelestialBodiesShown = sky.CelestialBodiesShown,
        }
    end
end

-- restores only captured VALUES (lighting props + pre-existing Atmosphere/Sky)
local function restoreLightingValues()
    if AuraX.lightSaved then
        for _, key in ipairs(LIGHT_PROPS) do
            pcall(function() Lighting[key] = AuraX.lightSaved[key] end)
        end
    end
    local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmo and savedAtmo then
        for key, value in pairs(savedAtmo) do
            pcall(function() atmo[key] = value end)
        end
    end
    local sky = Lighting:FindFirstChildOfClass("Sky")
    if sky and savedSky then
        for key, value in pairs(savedSky) do
            pcall(function() sky[key] = value end)
        end
    end
end

-- full unload restore: values + remove every instance AuraX added to Lighting
local function restoreLighting()
    restoreLightingValues()
    for _, inst in ipairs(AuraX.tracked) do
        if inst and inst.Parent == Lighting then
            pcall(function() inst:Destroy() end)
        end
    end
end

local function hoverify(btn, base, hovered)
    btn.MouseEnter:Connect(function()
        tween(btn, 0.12, { BackgroundColor3 = hovered })
    end)
    btn.MouseLeave:Connect(function()
        tween(btn, 0.18, { BackgroundColor3 = base })
    end)
end

function AuraX.setOn(self, on)
    if on then
        if not self.enabled then
            self.enabled = true
            if self.enable then pcall(self.enable, self) end
        end
    else
        if self.enabled then
            self.enabled = false
            if self.disable then pcall(self.disable, self) end
        end
    end
end

local function define(feature)
    feature.enabled = false
    AuraX.features[feature.id] = feature
    table.insert(AuraX.order, feature.id)
end

local function addToggle(parent, text, default, callback)
    local row = makeRow(parent, 32)
    rowLabel(row, text)
    local state = default == true
    local pill = new("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -2, 0.5, 0),
        Size = UDim2.new(0, 42, 0, 22),
        BackgroundColor3 = state and Theme.Accent or Theme.ToggleOff,
        BorderSizePixel = 0,
        Parent = row,
    })
    corner(pill, 11)
    local pillStroke = stroke(pill, Theme.Stroke, 1, 0.94)
    local knob = new("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = state and UDim2.new(1, -19, 0.5, 0) or UDim2.new(0, 3, 0.5, 0),
        Size = UDim2.new(0, 16, 0, 16),
        BackgroundColor3 = state and Theme.Knob or Theme.SubText,
        BorderSizePixel = 0,
        Parent = pill,
    })
    corner(knob, 8)

    local function set(on)
        state = on
        tween(pill, 0.2, { BackgroundColor3 = on and Theme.Accent or Theme.ToggleOff })
        tween(knob, 0.2, { Position = on and UDim2.new(1, -19, 0.5, 0) or UDim2.new(0, 3, 0.5, 0) })
        tween(knob, 0.2, { BackgroundColor3 = on and Theme.Knob or Theme.SubText })
        pillStroke.Transparency = on and 0.7 or 0.94
        playSound(on and 1.3 or 0.9)
        if callback then callback(on) end
    end

    row.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            set(not state)
        end
    end)
    row.MouseEnter:Connect(function()
        tween(row, 0.1, { BackgroundTransparency = 0.94, BackgroundColor3 = Theme.Hover })
    end)
    row.MouseLeave:Connect(function()
        tween(row, 0.15, { BackgroundTransparency = 1 })
    end)
    return { Set = set, Get = function() return state end }
end

local function addSlider(parent, text, min, max, default, decimals, callback, suffix)
    local row = makeRow(parent, 44)
    local label = rowLabel(row, text)
    label.Size = UDim2.new(1, -70, 0, 18)
    local mult = 10 ^ (decimals or 0)

    local valueLabel = new("TextLabel", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -2, 0, 0),
        Size = UDim2.new(0, 64, 0, 18),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        TextSize = 12,
        TextColor3 = Theme.Accent2,
        TextXAlignment = Enum.TextXAlignment.Right,
        Parent = row,
    })

    local trackFrame = new("Frame", {
        Position = UDim2.new(0, 2, 0, 24),
        Size = UDim2.new(1, -6, 0, 6),
        BackgroundColor3 = Theme.ToggleOff,
        BorderSizePixel = 0,
        Parent = row,
    })
    corner(trackFrame, 3)
    local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = trackFrame })
    corner(fill, 3)
    gradient(fill, Theme.Accent, Theme.Accent2)
    local knob = new("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.new(0, 13, 0, 13),
        BackgroundColor3 = Theme.Knob,
        BorderSizePixel = 0,
        Parent = trackFrame,
    })
    corner(knob, 7)

    local value = default
    local dragging = false

    local function setValue(v, fire)
        v = math.clamp(math.floor(v * mult + 0.5) / mult, min, max)
        value = v
        local pct = (v - min) / (max - min)
        valueLabel.Text = string.format("%." .. (decimals or 0) .. "f", v) .. (suffix or "")
        tween(fill, 0.06, { Size = UDim2.new(pct, 0, 1, 0) })
        tween(knob, 0.06, { Position = UDim2.new(pct, 0, 0.5, 0) })
        if fire ~= false and callback then callback(v) end
    end

    local function setFromX(x, fire)
        local pct = math.clamp((x - trackFrame.AbsolutePosition.X) / math.max(trackFrame.AbsoluteSize.X, 1), 0, 1)
        setValue(min + (max - min) * pct, fire)
    end

    trackFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    bind(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end))
    bind(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    setValue(default, false)
    return { Get = function() return value end, Set = function(v) setValue(v, false) end }
end

local function addDropdown(parent, text, options, default, callback)
    local row = makeRow(parent, 32)
    rowLabel(row, text)
    local open = false
    local current = default or options[1]

    local btn = new("TextButton", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, 3),
        Size = UDim2.new(0, 150, 0, 26),
        BackgroundColor3 = Theme.Panel2,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        Text = tostring(current),
        TextSize = 12,
        TextColor3 = Theme.Text,
        TextTruncate = Enum.TextTruncate.AtEnd,
        AutoButtonColor = false,
        Parent = row,
    })
    corner(btn, 7)
    local bs = stroke(btn, Theme.Stroke, 1, 0.93)
    local chevron = new("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -10, 0.5, 0),
        Size = UDim2.new(0, 8, 0, 8),
        BackgroundTransparency = 1,
        Parent = btn,
    })
    new("Frame", { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.3, 0, 0.5, 1), Size = UDim2.new(0, 7, 0, 1.6), Rotation = 45, BackgroundColor3 = Theme.SubText, BorderSizePixel = 0, Parent = chevron })
    new("Frame", { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.7, 0, 0.5, 1), Size = UDim2.new(0, 7, 0, 1.6), Rotation = -45, BackgroundColor3 = Theme.SubText, BorderSizePixel = 0, Parent = chevron })

    local list = new("Frame", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, 32),
        Size = UDim2.new(0, 150, 0, #options * 25),
        BackgroundColor3 = Theme.Panel2,
        BorderSizePixel = 0,
        Visible = false,
        ZIndex = 5,
        Parent = row,
    })
    corner(list, 8)
    stroke(list, Theme.Stroke, 1, 0.9)
    new("UIListLayout", { Padding = UDim.new(0, 0), Parent = list })

    for i, option in ipairs(options) do
        local optBtn = new("TextButton", {
            Size = UDim2.new(1, 0, 0, 25),
            BackgroundColor3 = Theme.Panel2,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            Font = Enum.Font.Gotham,
            Text = tostring(option),
            TextSize = 12,
            TextColor3 = option == current and Theme.Accent2 or Theme.Text,
            ZIndex = 6,
            AutoButtonColor = false,
            Parent = list,
        })
        if option == current then
            optBtn.BackgroundColor3 = Theme.Hover
            optBtn.BackgroundTransparency = 0.4
        end
        optBtn.MouseEnter:Connect(function()
            tween(optBtn, 0.1, { BackgroundTransparency = 0.4, BackgroundColor3 = Theme.Hover })
        end)
        optBtn.MouseLeave:Connect(function()
            if option ~= current then tween(optBtn, 0.15, { BackgroundTransparency = 1 }) end
        end)
        optBtn.MouseButton1Click:Connect(function()
            current = option
            btn.Text = tostring(option)
            for _, other in ipairs(list:GetChildren()) do
                if other:IsA("TextButton") then
                    other.TextColor3 = Theme.Text
                    other.BackgroundTransparency = 1
                end
            end
            optBtn.TextColor3 = Theme.Accent2
            optBtn.BackgroundTransparency = 0.4
            open = false
            row.ZIndex = 1
            list.Visible = false
            playSound(1.1)
            if callback then callback(option) end
        end)
    end

    btn.MouseButton1Click:Connect(function()
        open = not open
        list.Visible = open
        row.ZIndex = open and 25 or 1
        tween(chevron, 0.2, { Rotation = open and 180 or 0 })
        playSound(1)
    end)

    return { Get = function() return current end }
end

local function addColor(parent, text, default, callback)
    local row = makeRow(parent, 32)
    rowLabel(row, text)
    local open = false
    local h, s, v = Color3.toHSV(default)

    local swatch = new("TextButton", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, 3),
        Size = UDim2.new(0, 26, 0, 26),
        BackgroundColor3 = default,
        Text = "",
        AutoButtonColor = false,
        BorderSizePixel = 0,
        Parent = row,
    })
    corner(swatch, 7)
    stroke(swatch, Theme.Stroke, 1, 0.85)

    -- panel with preset swatches + HSV sliders
    local panel = new("Frame", {
        Position = UDim2.new(0, 0, 0, 32),
        Size = UDim2.new(1, 0, 0, 138),
        BackgroundColor3 = Theme.Panel2,
        BorderSizePixel = 0,
        Visible = false,
        ZIndex = 5,
        Parent = row,
    })
    corner(panel, 8)
    stroke(panel, Theme.Stroke, 1, 0.9)

    -- control refs (declared early so preset buttons can capture them)
    local hueCtl, satCtl, valCtl
    local satGrad, valGrad

    local function applyColor(fire)
        local c = Color3.fromHSV(h, s, v)
        swatch.BackgroundColor3 = c
        if satGrad then satGrad.Color = ColorSequence.new(Color3.new(1, 1, 1), Color3.fromHSV(h, 1, 1)) end
        if valGrad then valGrad.Color = ColorSequence.new(Color3.fromHSV(h, 1, 1), Color3.new(0, 0, 0)) end
        if fire and callback then callback(c) end
    end

    local function miniSlider(yPos, init, cb)
        local holder = new("Frame", {
            Position = UDim2.new(0, 10, 0, yPos),
            Size = UDim2.new(1, -20, 0, 14),
            BackgroundTransparency = 1,
            ZIndex = 6,
            Parent = panel,
        })
        local bar = new("Frame", {
            Position = UDim2.new(0, 0, 0.5, -2.5),
            Size = UDim2.new(1, 0, 0, 5),
            BackgroundColor3 = Color3.fromRGB(255, 255, 255),
            BorderSizePixel = 0,
            ZIndex = 6,
            Parent = holder,
        })
        corner(bar, 2)
        local grad = gradient(bar, Color3.new(1, 1, 1), Color3.new(0, 0, 0))
        local knob = new("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.new(init or 0, 0, 0.5, 0),
            Size = UDim2.new(0, 11, 0, 11),
            BackgroundColor3 = Theme.Knob,
            BorderSizePixel = 0,
            ZIndex = 7,
            Parent = holder,
        })
        corner(knob, 5)
        stroke(knob, Theme.Stroke, 1.4, 0.3)

        local dragging = false
        local value = init or 0

        local function setFromX(x, fire)
            local pct = math.clamp((x - holder.AbsolutePosition.X) / math.max(holder.AbsoluteSize.X, 1), 0, 1)
            value = pct
            knob.Position = UDim2.new(pct, 0, 0.5, 0)
            if fire ~= false and cb then cb(pct) end
        end

        bar.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                setFromX(input.Position.X)
            end
        end)
        bind(UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                setFromX(input.Position.X)
            end
        end))
        bind(UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end))

        return {
            Get = function() return value end,
            Set = function(nv, fire)
                value = math.clamp(nv, 0, 1)
                knob.Position = UDim2.new(value, 0, 0.5, 0)
                if fire ~= false and cb then cb(value) end
            end,
        }, grad
    end

    -- preset swatches
    local presets = {
        Color3.fromRGB(255, 255, 255), Color3.fromRGB(255, 90, 90),  Color3.fromRGB(255, 150, 60), Color3.fromRGB(255, 220, 80),
        Color3.fromRGB(90, 255, 140),  Color3.fromRGB(60, 255, 220), Color3.fromRGB(70, 190, 255), Color3.fromRGB(90, 120, 255),
        Color3.fromRGB(160, 90, 255),  Color3.fromRGB(255, 90, 220), Color3.fromRGB(255, 120, 170), Color3.fromRGB(24, 26, 34),
    }
    local grid = new("Frame", {
        Position = UDim2.new(0, 10, 0, 10),
        Size = UDim2.new(1, -20, 0, 54),
        BackgroundTransparency = 1,
        ZIndex = 6,
        Parent = panel,
    })
    new("UIGridLayout", { CellSize = UDim2.new(0, 24, 0, 24), CellPadding = UDim2.new(0, 6, 0, 6), Parent = grid })
    for _, presetColor in ipairs(presets) do
        local sw = new("TextButton", {
            Size = UDim2.new(0, 24, 0, 24),
            BackgroundColor3 = presetColor,
            Text = "",
            AutoButtonColor = false,
            BorderSizePixel = 0,
            ZIndex = 7,
            Parent = grid,
        })
        corner(sw, 6)
        sw.MouseButton1Click:Connect(function()
            h, s, v = Color3.toHSV(presetColor)
            hueCtl.Set(h, true)
            satCtl.Set(s, true)
            valCtl.Set(v, true)
            applyColor(true)
            playSound(1.2)
        end)
    end

    -- HSV mini sliders (H, S, V)
    local hueRef, hueGrad = miniSlider(72, h, function(nh)
        h = nh
        applyColor(true)
    end)
    hueCtl = hueRef
    hueGrad.Color = rainbowSequence()

    local satRef, sg = miniSlider(94, s, function(ns)
        s = ns
        applyColor(true)
    end)
    satCtl = satRef
    satGrad = sg

    local valRef, vg = miniSlider(116, v, function(nv)
        v = nv
        applyColor(true)
    end)
    valCtl = valRef
    valGrad = vg

    swatch.MouseButton1Click:Connect(function()
        open = not open
        panel.Visible = open
        row.ZIndex = open and 25 or 1
        tween(row, 0.22, { Size = open and UDim2.new(1, 0, 0, 176) or UDim2.new(1, 0, 0, 32) })
        playSound(1)
    end)

    applyColor(false)
    return { GetColor = function() return Color3.fromHSV(h, s, v) end }
end

local function ensureAtmosphere()
    local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
    if not atmo then
        atmo = track(new("Atmosphere", { Name = "AuraX_Atmosphere", Parent = Lighting }))
    end
    return atmo
end

local function ensureSky()
    local sky = Lighting:FindFirstChildOfClass("Sky")
    if not sky then
        sky = track(new("Sky", { Name = "AuraX_Sky", Parent = Lighting }))
    end
    return sky
end

local function applySkyTheme(name)
    local preset = SKY_PRESETS[name]
    if not preset then return end

    local atmo = ensureAtmosphere()
    atmo.Density = preset.atmo.density
    atmo.Color = preset.atmo.color
    atmo.Decay = preset.atmo.decay
    atmo.Glare = preset.atmo.glare
    atmo.Haze = preset.atmo.haze

    local sky = ensureSky()
    sky.StarCount = preset.sky.stars
    sky.SunAngularSize = preset.sky.sun
    sky.MoonAngularSize = preset.sky.moon
    sky.CelestialBodiesShown = preset.sky.cbs
    if SKYBOX_TEXTURES then
        pcall(function()
            sky.SkyboxBk = SKYBOX_TEXTURES.Bk
            sky.SkyboxDn = SKYBOX_TEXTURES.Dn
            sky.SkyboxFt = SKYBOX_TEXTURES.Ft
            sky.SkyboxLf = SKYBOX_TEXTURES.Lf
            sky.SkyboxRt = SKYBOX_TEXTURES.Rt
            sky.SkyboxUp = SKYBOX_TEXTURES.Up
        end)
    end

    Lighting.ClockTime = preset.clock
    Lighting.Brightness = preset.brightness
    Lighting.Ambient = preset.ambient
    Lighting.OutdoorAmbient = preset.outdoor

    local cc = Lighting:FindFirstChild("AuraX_ThemeCC")
    if not cc then
        cc = track(new("ColorCorrectionEffect", { Name = "AuraX_ThemeCC", Parent = Lighting }))
    end
    cc.TintColor = preset.cc.tint
    cc.Saturation = preset.cc.sat
    cc.Contrast = preset.cc.con
    cc.Brightness = preset.cc.bright
end

local function clearSkyTheme()
    -- only removes sky-theme instances; never touches other active effects
    local cc = Lighting:FindFirstChild("AuraX_ThemeCC")
    if cc then cc:Destroy() end
    local themedAtmo = Lighting:FindFirstChild("AuraX_Atmosphere")
    if themedAtmo then themedAtmo:Destroy() end
    local themedSky = Lighting:FindFirstChild("AuraX_Sky")
    if themedSky then themedSky:Destroy() end
    restoreLightingValues()
end

define({
    id = "WideView",
    tab = "Visuals",
    section = "Screen & Post FX",
    st = { fov = 104, saved = nil },
    enable = function(self)
        local cam = WorkspaceService.CurrentCamera
        if not cam then return end
        self.st.saved = { mode = cam.FieldOfViewMode, fov = cam.FieldOfView }
        cam.FieldOfViewMode = Enum.FieldOfViewMode.Horizontal
        cam.FieldOfView = self.st.fov
    end,
    disable = function(self)
        local cam = WorkspaceService.CurrentCamera
        if cam and self.st.saved then
            cam.FieldOfViewMode = self.st.saved.mode
            cam.FieldOfView = self.st.saved.fov
            self.st.saved = nil
        end
    end,
    build = function(self, rows)

define({
    id = "Movement",
    tab = "Movement",
    section = "Locomotion Polish",
    always = true,
    st = { animSpeed = 1, lean = false, leanAmount = 1, motor = nil, resetLean = false },
    onCharacter = function(self, char)
        local motor = nil
        local lower = char:FindFirstChild("LowerTorso")
        if lower then motor = lower:FindFirstChild("Root") end
        if not motor then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then motor = hrp:FindFirstChild("RootJoint") end
        end
        self.st.motor = motor
    end,
    update = function(self)
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        -- animation speed smoothing
        if hum and self.st.animSpeed ~= 1 then
            for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
                if math.abs(track.AdjustSpeed - self.st.animSpeed) > 0.02 then
                    track.AdjustSpeed = self.st.animSpeed
                end
            end
        end
        -- smooth lean into movement direction (Motor6D.Transform overlay)
        local motor = self.st.motor
        if motor and motor.Parent then
            if self.st.lean and char then
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    local lv = root.CFrame:VectorToObjectSpace(root.AssemblyLinearVelocity)
                    local amt = self.st.leanAmount * 0.45
                    local pitch = math.rad(math.clamp(lv.Z * amt, -12, 12))
                    local roll = math.rad(math.clamp(-lv.X * amt, -10, 10))
                    motor.Transform = CFrame.Angles(pitch, 0, roll)
                end
            elseif self.st.resetLean then
                motor.Transform = CFrame.new()
                self.st.resetLean = false
            end
        end
    end,
    build = function(self, rows)
        addSlider(rows, "Animation Speed", 0.3, 2, 1, 2, function(v)
            self.st.animSpeed = v
        end, "x")
        addToggle(rows, "Smooth Lean", false, function(on)
            self.st.lean = on
            if not on then self.st.resetLean = true end
        end)
        addSlider(rows, "Lean Intensity", 0.2, 2, 1, 2, function(v)
            self.st.leanAmount = v
        end, "x")
    end,
})

        rightFlat = rightFlat.Unit
        local pos = hit.Position + up * 0.04 + rightFlat * side * 0.35
        local p = new("Part", {
            Shape = Enum.PartType.Cylinder,

            CanTouch = false,
            CastShadow = false,
            Transparency = 0.25,
            Parent = fxRoot,
        })

        task.delay(1.15, function()
            if p.Parent then p:Destroy() end
        end)

        if self.st.emitter then self.st.emitter.Color = ColorSequence.new(c) end

        end

        em:Emit(90)

        tween(flash, 0.32, { Size = Vector3.new(3.4, 3.4, 3.4), Transparency = 1 }, Enum.EasingStyle.Quint)

                Shape = Enum.PartType.Ball,
                Size = Vector3.new(0.12, 0.12, 0.12),
                CFrame = CFrame.new(pos),

                Color = c,
                Anchored = false,
                CanCollide = false,
                CanQuery = false,
                CanTouch = false,
                CastShadow = false,
                Parent = fxRoot,
            })

            local a1 = new("Attachment", { Position = Vector3.new(0, -0.5, 0), Parent = sp })
            new("Trail", {
                Attachment0 = a0,
                Attachment1 = a1,
                Lifetime = 0.35,
                Transparency = NumberSequence.new(0.2, 1),
                LightEmission = 1,
                FaceCamera = true,
                Color = ColorSequence.new(c),
                Parent = sp,
            })

            if dir.Magnitude < 1 then dir = Vector3.new(0, 1, 0) end
            sp.AssemblyLinearVelocity = dir.Unit * math.random(11, 20)

                if sp.Parent then sp:Destroy() end
            end)
        end

            if anchor.Parent then anchor:Destroy() end
            if flash.Parent then flash:Destroy() end
        end)

        self.st.auto = on
        if self.st.autoThread then
            pcall(function() task.cancel(self.st.autoThread) end)
            self.st.autoThread = nil
        end
        if on and self.enabled then
            self.st.autoThread = task.spawn(function()
                while self.enabled and AuraX.alive do
                    local char = LocalPlayer.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")

                        local off = Vector3.new(math.random(-18, 18), math.random(8, 22), math.random(-18, 18))
                        self:burst(root.Position + off)
                    end
                    local waitTime = 1.2 + math.random() * 1.6
                    local elapsed = 0

                        elapsed = elapsed + task.wait(0.1)
                    end
                end

        self.st.conn = UserInputService.InputBegan:Connect(function(input, processed)
            if processed then return end

                return
            end
            local cam = WorkspaceService.CurrentCamera
            if not cam then return end
            local m = UserInputService:GetMouseLocation()
            local ray = cam:ViewportPointToRay(m.X, m.Y)
            local params = RaycastParams.new()
            local char = LocalPlayer.Character
            params.FilterDescendantsInstances = char and { char, fxRoot } or { fxRoot }
            params.FilterType = Enum.RaycastFilterType.Exclude
            local hit = WorkspaceService:Raycast(ray.Origin, ray.Direction * 600, params)
            local pos = hit and hit.Position or (ray.Origin + ray.Direction * 220)
            self:burst(pos + (hit and hit.Normal * 0.4 or Vector3.new(0, 0, 0)))
        end)
        self:setAuto(self.st.auto)
    end,
    disable = function(self)
        if self.st.conn then
            pcall(function() self.st.conn:Disconnect() end)
            self.st.conn = nil
        end
        if self.st.autoThread then
            pcall(function() task.cancel(self.st.autoThread) end)
            self.st.autoThread = nil
        end
    end,
    build = function(self, rows)
        addToggle(rows, "Click Fireworks", false, function(on)
            AuraX.setOn(self, on)
        end)
        addToggle(rows, "Auto Fireworks", false, function(on)
            self:setAuto(on)
        end)
        addToggle(rows, "Random Colors", true, function(on)
            self.st.random = on
        end)
        addColor(rows, "Burst Color", Color3.fromRGB(255, 120, 180), function(c)
            self.st.color = c
        end)
    end,
})

-- keeps the window reachable after drags / orientation changes
local function clampIntoView()
    local cam = WorkspaceService.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    local halfW = main.AbsoluteSize.X * 0.5
    local halfH = main.AbsoluteSize.Y * 0.5
    local minX = -(vp.X * 0.5 - math.min(halfW + 10, 90))
    local maxX = vp.X * 0.5 - math.min(halfW + 10, 90)
    local minY = -(vp.Y * 0.5 - 46)
    local maxY = vp.Y * 0.5 - math.min(halfH + 10, 70)
    main.Position = UDim2.new(
        0.5, math.clamp(main.Position.X.Offset, minX, maxX),
        0.5, math.clamp(main.Position.Y.Offset, minY, maxY)
    )
end

-- responsive scaling: the window shrinks gracefully on phones / tablets
local function fitToViewport()
    local cam = WorkspaceService.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
    AuraX.fitScale = math.clamp(math.min(vp.X / 680, vp.Y / 490), 0.55, 1)
    if AuraX.hubVisible then
        tween(uiScale, 0.2, { Scale = AuraX.fitScale })
    end
end
fitToViewport()
do
    local cam = WorkspaceService.CurrentCamera
    if cam then
        bind(cam:GetPropertyChangedSignal("ViewportSize"):Connect(fitToViewport))
    end
end

-- ===================================================================
-- RENDER DISPATCHER (single loop drives every effect + FPS meter)
-- ===================================================================
local clock = 0
local fpsFrames = 0
local fpsTime = 0

RunService:BindToRenderStep("AuraX_Update", Enum.RenderPriority.Character.Value + 2, function(dt)
    clock = clock + dt
    for _, id in ipairs(AuraX.order) do
        local f = AuraX.features[id]
        if f and (f.enabled or f.always) and f.update then
            local ok, err = pcall(f.update, f, dt, clock)
            if not ok then
                f.update = nil -- stop the broken updater instead of spamming errors
                warn("[AuraX] updater error in " .. id .. ": " .. tostring(err))
            end
        end
    end
    fpsFrames = fpsFrames + 1
    fpsTime = fpsTime + dt
    if fpsTime >= 0.5 then
        fpsLabel.Text = tostring(math.floor(fpsFrames / fpsTime + 0.5)) .. " FPS"
        fpsFrames = 0
        fpsTime = 0
    end
end)

-- ===================================================================
-- UNLOAD (full cleanup: effects, lighting, UI, connections)
-- ===================================================================
unload = function()
    if not AuraX.alive then return end
    AuraX.alive = false
    for _, id in ipairs(AuraX.order) do
        local f = AuraX.features[id]
        if f and f.enabled then
            if f.disable then pcall(f.disable, f) end
            f.enabled = false
        end
    end
    pcall(function() RunService:UnbindFromRenderStep("AuraX_Update") end)
    for _, conn in ipairs(AuraX.connections) do
        pcall(function() conn:Disconnect() end)
    end
    pcall(function() UserInputService.MouseIconEnabled = true end)
    pcall(restoreLighting)
    local cursorFeature = AuraX.features.Cursor
    if cursorFeature and cursorFeature.gui then
        pcall(function() cursorFeature.gui:Destroy() end)
    end
    pcall(function() fxRoot:Destroy() end)
    for _, inst in ipairs(AuraX.tracked) do
        pcall(function()
            if inst then inst:Destroy() end
        end)
    end
    pcall(function() guiHub:Destroy() end)
end
